18//! Background Bitswap service.
19//!
20//! The role of Bitswap service is to handle Bitswap RPC requests, specifically
21//! `bitswap_unstable_get(cid)` and `bitswap_unstable_stream(cids)`.
22//!
23//! In order to handle a request for a Bitswap block with a given CID, [`BitswapService`] issues
24//! Bitswap "have" request to all the connected Bitswap peers, then issues Bitswap "block" request
25//! to the first peer that answered "yes" to that request.
26//!
27//! Note that we have [`BitswapService`] per chain, even though currently
28//! [`NetworkService`](crate::network_service::NetworkService) doesn't track what chain the Bitswap
29//! request is destined to, and doesn't track what chain peers responded to it to forward the
30//! response to specific chain's [`BitswapService`]. As a result, [`BitswapService`] receives the
31//! responses intended for all the other Bitswap services as well. This should be fixed in
32//! [`NetworkService`](crate::network_service::NetworkService), but it is somewhat mitigated in
33//! [`BitswapService`] by not decoding the incoming Bitswap messages when there are no active
34//! requests.
35//
36// TODO: backpressure.
37//
38// TODO: wait a bit longer after receiving the first "have" response and randomly select the peer
39// for "block" request to distribute the load.
40//
41// TODO: do we need to retry the request with another peer if the first one didn't return the data
42// within the given time?
43//
44// TODO: do we need to ban peers that do not return the requested data after answering "yes" to
45// a "have" request? How much to wait for a response?
46//
47// TODO: do we need a "reputation system" to prefer peers that respond faster then others?

165    /// Request a Bitswap block.
166    pub async fn bitswap_get(&self, cid: String) -> Result<Vec<u8>, BitswapGetError> {
167        // Decoding CID is fast, so we can fail early on the API user side.
168        let cid = Cid::from_str(&cid).map_err(BitswapGetError::InvalidCid)?;
169
170        let (result_tx, result_rx) = oneshot::channel();
171
172        self.messages_tx
173            .send(ToBackground::BitswapBlock { cid, result_tx })
174            .await
175            .unwrap();
176
177        result_rx.await.unwrap()
178    }
179
180    /// Subscribe to a stream of Bitswap blocks. On success, returns a [`BitswapStreamHandle`]
181    /// whose `events_rx` yields one `(cid_string, BlockResult)` event per input CID, in arrival
182    /// order (the order in which each CID resolves), not input order.
183    ///
184    /// Top-level errors:
185    /// - `-32801 TooManyCids`, `-32802 EmptyCids`, `-32803 DuplicateCids`: batch-input
186    ///   validation, surfaced before any subscription state is created.
187    /// - `-32812 FailRetryBackoff`: wholesale Have-broadcast failure at subscription-open time
188    ///   (no Bitswap peers connected, or network send queue full).
189    ///
190    /// Per-CID failures observed *after* the subscription has opened (e.g. all peers
191    /// disconnect mid-stream, individual `Block` request errors) still surface as
192    /// `streamItemError` events, as per the spec.
193    ///
194    /// Dropping the returned handle (explicit unsubscribe or client disconnect) cancels remaining
195    /// work and emits a Bitswap Cancel wantlist to peers we previously contacted.
196    pub async fn bitswap_stream(
197        &self,
198        cids: Vec<String>,
199    ) -> Result<BitswapStreamHandle, BitswapGetError> {
200        let entries = parse_and_dedup(cids)?;
201
202        // Channel size matches typical batch sizes; events_rx is drained promptly by the JSON-RPC
203        // layer so back-pressure here is unlikely.
204        let (events_tx, events_rx) = async_channel::bounded(MAX_CIDS_PER_REQUEST);
205        let (ready_tx, ready_rx) = oneshot::channel();
206
207        self.messages_tx
208            .send(ToBackground::BitswapBatch {
209                entries,
210                events_tx,
211                ready_tx,
212            })
213            .await
214            .unwrap();
215
216        let batch_id = ready_rx.await.unwrap()?;
217
218        Ok(BitswapStreamHandle {
219            events_rx,
220            _cancel_guard: BatchCancelGuard {
221                batch_id,
222                messages_tx: self.messages_tx.clone(),
223            },
224        })
225    }

238/// Active subscription handle for [`BitswapService::bitswap_stream`].
239///
240/// `events_rx` yields one event per input CID in arrival order. After all events have been
241/// emitted, the channel closes naturally. Dropping the handle before then signals the service to
242/// abort remaining work and emit a Bitswap Cancel wantlist for in-flight CIDs.

244    /// Receiver of per-CID events. `(cid_string, BlockResult)` per spec.

246    _cancel_guard: BatchCancelGuard,
247}

249/// Internal RAII guard that fires `ToBackground::CancelBatch` on drop. Held by
250/// [`BitswapStreamHandle`] (covers explicit unsubscribe and client disconnect).
251struct BatchCancelGuard {
252    batch_id: BatchId,
253    messages_tx: async_channel::Sender<ToBackground>,
254}
255
256impl Drop for BatchCancelGuard {
257    fn drop(&mut self) {
258        // Best-effort. If the service's channel is closed (service shut down already) or full
259        // (extremely unlikely — bounded(32)), we have no recourse since Drop can't await.
260        // A no-op CancelBatch on an already-finished batch is harmless: the service will look up
261        // the batch_id, find nothing, and ignore the message.
262        let _ = self.messages_tx.try_send(ToBackground::CancelBatch {
263            batch_id: self.batch_id,
264        });
265    }
266}

268/// Error by [`BitswapService::bitswap_get`].
269#[derive(Debug, derive_more::Display, derive_more::Error, Clone)]
270pub enum BitswapGetError {
271    /// Invalid/unsupported CID.
272    #[display("Invalid CID: {_0}")]
273    InvalidCid(cid::ParseError),
274    /// No Bitswap peers connected, can't issue "have" request.
275    #[display("No Bitswap peers connected, can't issue \"have\" request.")]
276    NoPeers,
277    /// "Block" request to selected peer failed after successful "have" request.
278    #[display("\"Block\" request to selected peer failed after successful \"have\" request.")]
279    BlockRequestFailed,
280    /// Network sending queue is full.
281    #[display("Network sending queue is full.")]
282    QueueFull,
283    /// Requested CID not found.
284    #[display("No connected peers have the CID requested.")]
285    NotFound,
286    /// Request timeout.
287    #[display("Request timeout.")]
288    Timeout,
289    /// Too many CIDs in a single batch request.
290    #[display("Too many CIDs in batch request: max {max}, got {got}.")]
291    TooManyCids {
292        /// Configured limit.
293        max: usize,
294        /// Number of CIDs in the rejected request.
295        got: usize,
296    },
297    /// Input array is empty.
298    #[display("Input array is empty.")]
299    EmptyCids,
300    /// Same CID appears more than once in the input. Two-stage detection: literal-string match,
301    /// or two distinct strings decoding to the same content digest.
302    #[display("Input contains duplicate CIDs.")]
303    DuplicateCids,
304}

306/// JSON-RPC error codes used by the `bitswap_*` namespace.
307///
308/// Clients should use the numeric code to determine recovery action, not parse the
309/// human-readable message string. Top-level batch-input validation codes
310/// (`TooManyCids`, `EmptyCids`, `DuplicateCids`) only appear as top-level errors of
311/// `bitswap_unstable_stream`. Per-request/per-event codes (`Fail`, `FailRetry`,
312/// `FailRetryBackoff`) appear both as the top-level error of `bitswap_unstable_get`
313/// and inside `streamItemError` events.
314enum BitswapJsonRpcError {
315    /// Batch-input validation: `cids.len()` exceeds the implementation maximum.
316    TooManyCids = -32801,
317    /// Batch-input validation: `cids` is empty.
318    EmptyCids = -32802,
319    /// Batch-input validation: duplicate CID in input.
320    DuplicateCids = -32803,
321    /// Permanent failure for this request. E.g., there is no requested data in the network.
322    /// Doesn't make sense to retry until you put the data on chain.
323    Fail = -32810,
324    /// Transient failure. Can retry immediately.
325    ///
326    /// Even though the client can retry immediately, the clients are encouraged to rate-limit the
327    /// retry attempts and retry count, e.g. introducing a delay of 50ms between retries.
328    FailRetry = -32811,
329    /// Transient failure. Retry after a backoff delay.
330    ///
331    /// The recommended backoff delay is 5s.
332    FailRetryBackoff = -32812,
333}

335impl BitswapGetError {
336    /// Build a complete JSON-RPC error response string for this error.
337    pub fn to_json_rpc_error(&self, request_id_json: &str) -> String {
338        let message = self.to_string();
339
340        let error_response = match self {
341            BitswapGetError::InvalidCid(_) => parse::ErrorResponse::InvalidParams(Some(&message)),
342            BitswapGetError::NotFound => {
343                parse::ErrorResponse::ApplicationDefined(BitswapJsonRpcError::Fail as i64, &message)
344            }
345            BitswapGetError::BlockRequestFailed | BitswapGetError::Timeout => {
346                parse::ErrorResponse::ApplicationDefined(
347                    BitswapJsonRpcError::FailRetry as i64,
348                    &message,
349                )
350            }
351            BitswapGetError::QueueFull | BitswapGetError::NoPeers => {
352                parse::ErrorResponse::ApplicationDefined(
353                    BitswapJsonRpcError::FailRetryBackoff as i64,
354                    &message,
355                )
356            }
357            BitswapGetError::TooManyCids { .. } => parse::ErrorResponse::ApplicationDefined(
358                BitswapJsonRpcError::TooManyCids as i64,
359                &message,
360            ),
361            BitswapGetError::EmptyCids => parse::ErrorResponse::ApplicationDefined(
362                BitswapJsonRpcError::EmptyCids as i64,
363                &message,
364            ),
365            BitswapGetError::DuplicateCids => parse::ErrorResponse::ApplicationDefined(
366                BitswapJsonRpcError::DuplicateCids as i64,
367                &message,
368            ),
369        };
370
371        parse::build_error_response(request_id_json, error_response, None)
372    }
373
374    /// Returns the JSON-RPC `(code, message)` pair to embed inside a per-CID `streamItemError`
375    /// event of `bitswap_unstable_streamEvent`. The code uses the same retry categories as the
376    /// top-level error of `bitswap_unstable_get`, so callers can reuse retry logic.
377    ///
378    /// The top-level batch-validation variants (`TooManyCids`, `EmptyCids`, `DuplicateCids`)
379    /// never appear per-event — they reject the subscription before any event is emitted.
380    pub fn to_block_result_err(&self) -> (i32, String) {
381        const INVALID_PARAMS: i32 = -32602;
382        let code = match self {
383            BitswapGetError::InvalidCid(_) => INVALID_PARAMS,
384            BitswapGetError::NotFound => BitswapJsonRpcError::Fail as i32,
385            BitswapGetError::BlockRequestFailed | BitswapGetError::Timeout => {
386                BitswapJsonRpcError::FailRetry as i32
387            }
388            BitswapGetError::QueueFull | BitswapGetError::NoPeers => {
389                BitswapJsonRpcError::FailRetryBackoff as i32
390            }
391            BitswapGetError::TooManyCids { .. }
392            | BitswapGetError::EmptyCids
393            | BitswapGetError::DuplicateCids => {
394                unreachable!("top-level batch-validation error, never emitted per-CID")
395            }
396        };
397        (code, self.to_string())
398    }
399}

410/// Validates and de-duplicates the input CIDs of `bitswap_unstable_stream`.
411///
412/// On success returns one entry per input CID, in input order, preserving the original string and
413/// the parse result. Caller-side per-CID `Err(InvalidCid)` reporting is left to the JSON-RPC layer
414/// since the spec emits invalid CIDs as per-CID errors rather than aborting the whole call.
415///
416/// Failure cases (returned as `Err(_)` so the caller emits a top-level JSON-RPC error):
417/// * `TooManyCids` if the input exceeds [`MAX_CIDS_PER_REQUEST`].
418/// * `DuplicateCids` if two inputs are literally-equal strings, **or** if two valid-but-different
419///   strings decode to the same [`Cid`] (digest collision).
420pub fn parse_and_dedup(
421    cids: Vec<String>,
422) -> Result<Vec<(String, Result<Cid, cid::ParseError>)>, BitswapGetError> {
423    if cids.is_empty() {
424        return Err(BitswapGetError::EmptyCids);
425    }
426    if cids.len() > MAX_CIDS_PER_REQUEST {
427        return Err(BitswapGetError::TooManyCids {
428            max: MAX_CIDS_PER_REQUEST,
429            got: cids.len(),
430        });
431    }
432
433    let mut seen_strings: hashbrown::HashSet<String> =
434        hashbrown::HashSet::with_capacity(cids.len());
435    // Dedup by 32-byte content digest, matching the literal spec wording (and the polkadot-sdk
436    // reference impl). Two cosmetically different CID strings that decode to the same digest
437    // collide here, regardless of multicodec or multihash type.
438    let mut seen_digests: hashbrown::HashSet<[u8; 32]> =
439        hashbrown::HashSet::with_capacity(cids.len());
440    let mut out = Vec::with_capacity(cids.len());
441
442    for cid_str in cids {
443        if !seen_strings.insert(cid_str.clone()) {
444            return Err(BitswapGetError::DuplicateCids);
445        }
446
447        let parsed = Cid::from_str(&cid_str);
448        if let Ok(c) = &parsed {
449            if !seen_digests.insert(*c.digest()) {
450                return Err(BitswapGetError::DuplicateCids);
451            }
452        }
453
454        out.push((cid_str, parsed));
455    }
456
457    Ok(out)
458}

460enum ToBackground {
461    BitswapBlock {
462        cid: Cid,
463        result_tx: oneshot::Sender<Result<Vec<u8>, BitswapGetError>>,

466    /// and reports the outcome back via `ready_tx`: `Ok(BatchId)` once the broadcast has landed

478    /// Cancel an in-flight batch. Idempotent: if the batch already finished, this is a no-op.

518/// State for an in-flight batch. Allocated when `BitswapBatch` is received and removed once all
519/// slots have been resolved (or the batch is explicitly cancelled).

521    /// Sender for per-CID outcomes, in arrival order. Dropping it closes the channel, signalling

529    /// Peers we sent this batch's Have broadcast to. Used to address Cancel wantlist on

532    /// Number of slots still pending resolution. When this reaches zero, finalize the batch.

579    next_batch_id_inner: u64,
580    /// RNG.
581    randomness: rand_chacha::ChaCha20Rng,
582
583    // The fields below are populated if the broadcast of the "have" message was successfully
584    // forwarded to the network. Request is tracked from this moment until the requested data is
585    // received or the request time-outs.
586    //
587    /// All tracked requests.
588    requests: hashbrown::HashMap<RequestId, Request<TPlat>, fnv::FnvBuildHasher>,
589    /// Requests ordered by timeout.
590    requests_by_timeout: BTreeSet<(TPlat::Instant, RequestId)>,
591    /// Requests ordered by CID. The request IDs in the `VecDeque` are ordered by their timeout if
592    /// the platform implementation of `now` is monothonic (true for
593    /// [`crate::platform::DefaultPlatform`]).
594    requests_by_cid: hashbrown::HashMap<Cid, VecDeque<RequestId>, util::SipHasherBuild>,
595    /// In-flight batches. Each entry corresponds to a `bitswap_stream` call.
596    batches: hashbrown::HashMap<BatchId, Batch, fnv::FnvBuildHasher>,
597}
598
599impl<TPlat: PlatformRef> BackgroundTask<TPlat> {
600    fn allocate_request_id(&mut self) -> RequestId {
601        let request_id = RequestId(self.next_request_id_inner);
602        self.next_request_id_inner += 1;
603
604        request_id
605    }
606
607    fn allocate_batch_id(&mut self) -> BatchId {
608        let batch_id = BatchId(self.next_batch_id_inner);
609        self.next_batch_id_inner += 1;
610        batch_id
611    }
612
613    /// Final disposition of a single request slot. Routes the resolution either to the original
614    /// `bitswap_get` caller (`Single`) or to the owning batch (`Batch`).
615    fn deliver_slot(&mut self, slot_output: SlotOutput, result: Result<Vec<u8>, BitswapGetError>) {
616        match slot_output {
617            SlotOutput::Single(tx) => {
618                let _ = tx.send(result);
619            }
620            SlotOutput::Batch { batch_id, slot_idx } => {
621                self.deliver_batch_slot(batch_id, slot_idx, result);
622            }
623        }
624    }
625
626    /// Deliver one resolved slot to its owning batch and finalize the batch if all slots are now
627    /// resolved.
628    fn deliver_batch_slot(
629        &mut self,
630        batch_id: BatchId,
631        slot_idx: usize,
632        result: Result<Vec<u8>, BitswapGetError>,
633    ) {
634        let block_result = match result {
635            Ok(data) => BlockResult::Ok(data),
636            Err(e) => BlockResult::Err(e),
637        };
638
639        // Touch state in two passes so we can release the borrow before potentially calling
640        // `cancel_batch` (which also borrows `self.batches`).
641        let (should_cancel, should_finalize) = {
642            let Some(batch) = self.batches.get_mut(&batch_id) else {
643                // The batch was already finalized or cancelled; the resolution is stale.
644                return;
645            };
646
647            // Mark this slot as no longer in-flight regardless of mode.
648            batch.slots[slot_idx] = None;
649            batch.pending_count = batch.pending_count.saturating_sub(1);
650
651            let cid_str = batch.cid_strs[slot_idx].clone();
652            let mut should_cancel = false;
653
654            match batch.events_tx.try_send((cid_str, block_result)) {
655                Ok(()) => {}
656                Err(async_channel::TrySendError::Closed(_)) => {
657                    // Receiver dropped — JSON-RPC client disconnected or unsubscribed.
658                    // Cancel the rest of the batch and emit a Bitswap Cancel wantlist.
659                    should_cancel = true;
660                }
661                Err(async_channel::TrySendError::Full(_)) => {
662                    // Invariant: parse_and_dedup caps entries.len() at MAX_CIDS_PER_REQUEST
663                    // and events_tx is bounded(MAX_CIDS_PER_REQUEST). At most `total` items
664                    // are ever pushed for a batch, so the buffer cannot saturate.
665                    log!(
666                        &self.platform,
667                        Error,
668                        &self.log_target,
669                        "BUG: stream events channel full — invariant violated"
670                    );
671                    unreachable!()
672                }
673            }
674
675            let should_finalize = !should_cancel && batch.pending_count == 0;
676            (should_cancel, should_finalize)
677        };
678
679        if should_cancel {
680            self.cancel_batch(batch_id);
681        } else if should_finalize {
682            self.finalize_batch(batch_id);
683        }
684    }
685
686    /// Finalize a batch whose slots have all been resolved. Closes the events channel so the
687    /// JSON-RPC pump can emit the spec-required `streamDone` event.
688    fn finalize_batch(&mut self, batch_id: BatchId) {
689        let Some(batch) = self.batches.remove(&batch_id) else {
690            return;
691        };
692        debug_assert_eq!(batch.pending_count, 0);
693
694        log!(
695            &self.platform,
696            Trace,
697            &self.log_target,
698            "batch finalized",
699            batch_id = batch_id.0,
700            slots = batch.cid_strs.len()
701        );
702
703        // Dropping the sender closes the channel. The JSON-RPC pump task will see the channel
704        // close and end its loop after the last event has been delivered.
705        drop(batch.events_tx);
706    }
707
708    /// Cancel a batch: tear down its still-pending slots, then send a wire-level `Cancel(cid)`
709    /// for each CID this batch was the *last* local requester of. Idempotent.
710    ///
711    /// Only the last-reference case emits a Cancel — `requests_by_cid` is a multi-map and peers
712    /// track want-state per-CID, not per-requester, so an unconditional Cancel would starve any
713    /// concurrent batch or `bitswap_get` waiting on the same CID.
714    fn cancel_batch(&mut self, batch_id: BatchId) {
715        let Some(batch) = self.batches.remove(&batch_id) else {
716            return;
717        };
718
719        // Walk pending slots, evict their `RequestId`s from the global tracking maps, and gather
720        // the CIDs whose last local reference just went away (those are the ones safe to Cancel
721        // on the wire).
722        let mut pending_cids: Vec<Cid> = Vec::new();
723        for slot in batch.slots.into_iter() {
724            let Some(request_id) = slot else { continue };
725            let Some(request) = self.requests.remove(&request_id) else {
726                continue;
727            };
728            let _ = self
729                .requests_by_timeout
730                .remove(&(request.timeout, request_id));
731
732            if let hashbrown::hash_map::Entry::Occupied(mut entry) =
733                self.requests_by_cid.entry(request.cid.clone())
734            {
735                entry.get_mut().retain(|id| *id != request_id);
736                if entry.get().is_empty() {
737                    entry.remove();
738                    pending_cids.push(request.cid);
739                }
740            }
741        }
742
743        if pending_cids.is_empty() || batch.peers_for_cancel.is_empty() {
744            log!(
745                &self.platform,
746                Trace,
747                &self.log_target,
748                "batch cancelled (nothing to cancel on wire)",
749                batch_id = batch_id.0,
750                pending_cids = pending_cids.len(),
751                peers = batch.peers_for_cancel.len()
752            );
753            return;
754        }
755
756        log!(
757            &self.platform,
758            Trace,
759            &self.log_target,
760            "sending cancel wantlist",
761            batch_id = batch_id.0,
762            cids = ?pending_cids,
763            peers = batch.peers_for_cancel.len()
764        );
765
766        // One Cancel wantlist message containing all pending CIDs, sent to every peer this
767        // batch's Have broadcast reached. Substrate Bitswap servers don't track pending
768        // want-lists today, so Cancel is a no-op there; we send it for IPFS-spec conformance.
769        let message = build_bitswap_cancel_message(pending_cids.iter());
770
771        for peer in batch.peers_for_cancel {
772            let network_service = self.network_service.clone();
773            let msg = message.clone();
774            // Fire-and-forget. Cancel is best-effort and does not need to be awaited; if it
775            // fails the peer will eventually expire its want-list state on its own.
776            self.platform
777                .spawn_task(self.log_target.clone().into(), async move {
778                    let _ = network_service.send_bitswap_message(peer, msg).await;
779                });
780        }
781    }
782}

784fn bitswap_have_message(cid: &Cid) -> Vec<u8> {
785    build_bitswap_message(iter::once(cid), WantType::Have, true, false)
786}

788fn bitswap_block_message(cid: &Cid) -> Vec<u8> {
789    build_bitswap_message(iter::once(cid), WantType::Block, false, false)

792async fn background_task<TPlat: PlatformRef>(mut task: BackgroundTask<TPlat>) {
793    loop {
794        // Make sure to yield at every loop to provide better tasks granularity.
795        futures_lite::future::yield_now().await;
796
797        enum WakeUpReason {
798            MustSubscribeNetworkEvents,
799            NetworkEvent(network_service::BitswapEvent),
800            Message(ToBackground),
801            HaveBroadcastResult(HaveBroadcastResult),
802            BlockRequestResult((Result<(), SendBitswapMessageError>, Cid)),
803            RequestTimeout,
804            ForegroundClosed,
805        }
806
807        let wake_up_reason = {
808            let backpressure_messages = task.pending_have_broadcast.is_some();
809
810            async {
811                if let Some(from_network_service) = task.from_network_service.as_mut() {
812                    match from_network_service.next().await {
813                        Some(ev) => WakeUpReason::NetworkEvent(ev),
814                        None => {
815                            task.from_network_service = None;
816                            WakeUpReason::MustSubscribeNetworkEvents
817                        }
818                    }
819                } else {
820                    WakeUpReason::MustSubscribeNetworkEvents
821                }
822            }
823            .or(async {
824                if !backpressure_messages {
825                    task.messages_rx
826                        .next()
827                        .await
828                        .map_or(WakeUpReason::ForegroundClosed, WakeUpReason::Message)
829                } else {
830                    future::pending().await
831                }
832            })
833            .or(async {
834                if let Some(pending_have_broadcast) = &mut task.pending_have_broadcast {
835                    let result = pending_have_broadcast.await;
836                    task.pending_have_broadcast = None;
837                    WakeUpReason::HaveBroadcastResult(result)
838                } else {
839                    future::pending().await
840                }
841            })
842            .or(async {
843                if !task.pending_block_requests.is_empty() {
844                    let result = task
845                        .pending_block_requests
846                        .next()
847                        .await
848                        .expect("non-empty; qed");
849                    WakeUpReason::BlockRequestResult(result)
850                } else {
851                    future::pending().await
852                }
853            })
854            .or(async {
855                if let Some((first_timeout, _request_id)) = task.requests_by_timeout.first() {
856                    let now = task.platform.now();
857
858                    if now < *first_timeout {
859                        task.platform.sleep(first_timeout.clone() - now).await;
860                    }
861
862                    WakeUpReason::RequestTimeout
863                } else {
864                    future::pending().await
865                }
866            })
867            .await
868        };
869
870        // The handlers below are mostly in the order of a typical flow.
871        match wake_up_reason {
872            WakeUpReason::MustSubscribeNetworkEvents => {
873                debug_assert!(task.from_network_service.is_none());
874                task.from_network_service = Some(Box::pin(
875                    // As documented, `subscribe().await` is expected to return quickly.
876                    task.network_service.subscribe_bitswap().await,
877                ));
878            }
879            WakeUpReason::Message(ToBackground::BitswapBlock { cid, result_tx }) => {
880                debug_assert!(task.pending_have_broadcast.is_none());
881
882                log!(
883                    &task.platform,
884                    Trace,
885                    &task.log_target,
886                    "queueing have wantlist (single)",
887                    cid
888                );
889
890                let message = bitswap_have_message(&cid);
891                let network_service = task.network_service.clone();
892
893                // TODO: does it make sense to group the new request with the existing ones for the
894                //       same CID and don't actually broadcast the "have" request?
895
896                // Network service can be back-pressuring, so we run this in the background.
897                task.pending_have_broadcast = Some(Box::pin(async move {
898                    let result = network_service.broadcast_bitswap_message(message).await;
899                    (result, HaveContext::Single { cid, result_tx })
900                }));
901            }
902            WakeUpReason::Message(ToBackground::BitswapBatch {
903                entries,
904                events_tx,
905                ready_tx,
906            }) => {
907                debug_assert!(task.pending_have_broadcast.is_none());
908
909                // Allocate the batch up front. The BatchId is reported back via `ready_tx` only
910                // after the Have broadcast resolves (success or wholesale failure), so wholesale
911                // failures surface as top-level JSON-RPC errors instead of per-slot ones.
912                let batch_id = task.allocate_batch_id();
913
914                let total = entries.len();
915                let mut cid_strs: Vec<String> = Vec::with_capacity(total);
916                let mut slots: Vec<Option<RequestId>> = Vec::with_capacity(total);
917                // Slots whose CID parsed successfully — these will have their RequestId set
918                // after the Have broadcast lands. (slot_idx, cid).
919                let mut valid_cids: Vec<(usize, Cid)> = Vec::with_capacity(total);
920                // Slots whose CID failed to parse — pre-resolve as `InvalidCid` per spec.
921                // (cid_str, err).
922                let mut invalid_slots: Vec<(String, BitswapGetError)> = Vec::new();
923
924                for (slot_idx, (cid_str, parsed)) in entries.into_iter().enumerate() {
925                    cid_strs.push(cid_str.clone());
926                    slots.push(None);
927                    match parsed {
928                        Ok(c) => valid_cids.push((slot_idx, c)),
929                        Err(e) => {
930                            invalid_slots.push((cid_str, BitswapGetError::InvalidCid(e)));
931                        }
932                    }
933                }
934
935                let mut batch = Batch {
936                    events_tx,
937                    cid_strs,
938                    slots,
939                    peers_for_cancel: Vec::new(),
940                    pending_count: total,
941                };
942
943                // Pre-resolve invalid-CID slots by pushing them straight into the events channel.
944                // try_send cannot fail here: events_tx is bounded(MAX_CIDS_PER_REQUEST) and the
945                // total events ever pushed for this batch is bounded by entries.len(); the
946                // receiver is also held by BitswapStreamHandle so the channel cannot be Closed.
947                for (cid_str, err) in invalid_slots {
948                    batch.pending_count = batch.pending_count.saturating_sub(1);
949                    batch
950                        .events_tx
951                        .try_send((cid_str, BlockResult::Err(err)))
952                        .unwrap_or_else(|_| unreachable!());
953                }
954
955                log!(
956                    &task.platform,
957                    Trace,
958                    &task.log_target,
959                    "batch queued",
960                    batch_id = batch_id.0,
961                    total,
962                    valid = valid_cids.len(),
963                    invalid = total - valid_cids.len()
964                );
965
966                // Insert the batch before the broadcast: if the caller drops mid-await and a
967                // CancelBatch arrives, it must find an entry to cancel.
968                task.batches.insert(batch_id, batch);
969
970                if valid_cids.is_empty() {
971                    // No wire I/O needed. Signal Ok(batch_id) so the caller can construct its
972                    // cancel guard / stream handle, then finalize immediately (all-invalid case)
973                    // or leave the batch to await ordinary resolution paths.
974                    let _ = ready_tx.send(Ok(batch_id));
975                    if task
976                        .batches
977                        .get(&batch_id)
978                        .map_or(true, |b| b.pending_count == 0)
979                    {
980                        task.finalize_batch(batch_id);
981                    }
982                    continue;
983                }
984
985                log!(
986                    &task.platform,
987                    Trace,
988                    &task.log_target,
989                    "queueing have wantlist (batch)",
990                    batch_id = batch_id.0,
991                    cids = ?valid_cids.iter().map(|(_, c)| c).collect::<Vec<_>>()
992                );
993
994                let message = build_bitswap_message(
995                    valid_cids.iter().map(|(_, c)| c),
996                    WantType::Have,
997                    true,
998                    false,
999                );
1000                let network_service = task.network_service.clone();
1001                task.pending_have_broadcast = Some(Box::pin(async move {
1002                    let result = network_service.broadcast_bitswap_message(message).await;
1003                    (
1004                        result,
1005                        HaveContext::Batch {
1006                            batch_id,
1007                            cids: valid_cids,
1008                            ready_tx,
1009                        },
1010                    )
1011                }));
1012            }
1013            WakeUpReason::Message(ToBackground::CancelBatch { batch_id }) => {
1014                log!(
1015                    &task.platform,
1016                    Trace,
1017                    &task.log_target,
1018                    "cancel requested",
1019                    batch_id = batch_id.0
1020                );
1021                task.cancel_batch(batch_id);
1022            }
1023            WakeUpReason::HaveBroadcastResult((result, ctx)) => match ctx {
1024                HaveContext::Single { cid, result_tx } => {
1025                    let broadcast_to = match result {
1026                        Ok(peers) => peers,
1027                        Err(err) => {
1028                            log!(
1029                                &task.platform,
1030                                Trace,
1031                                &task.log_target,
1032                                "have broadcast failed (single)",
1033                                cid,
1034                                ?err
1035                            );
1036                            let _ = result_tx.send(Err(err.into()));
1037                            continue;
1038                        }
1039                    };
1040
1041                    log!(
1042                        &task.platform,
1043                        Trace,
1044                        &task.log_target,
1045                        "have broadcast sent (single)",
1046                        cid,
1047                        peers = broadcast_to.len()
1048                    );
1049
1050                    let request_id = task.allocate_request_id();
1051                    let timeout = task.platform.now() + Duration::from_secs(10);
1052
1053                    let have_peers = {
1054                        let mut have_peers = hashbrown::HashSet::with_capacity_and_hasher(
1055                            broadcast_to.len(),
1056                            util::SipHasherBuild::new({
1057                                let mut seed = [0; 16];
1058                                task.randomness.fill_bytes(&mut seed);
1059                                seed
1060                            }),
1061                        );
1062                        have_peers.extend(broadcast_to.into_iter());
1063                        have_peers
1064                    };
1065
1066                    task.requests.insert(
1067                        request_id,
1068                        Request {
1069                            result_tx: SlotOutput::Single(result_tx),
1070                            timeout: timeout.clone(),
1071                            stage: RequestStage::Have(have_peers),
1072                            cid: cid.clone(),
1073                        },
1074                    );
1075                    task.requests_by_timeout.insert((timeout, request_id));
1076                    task.requests_by_cid
1077                        .entry(cid)
1078                        .or_default()
1079                        .push_back(request_id);
1080                }
1081                HaveContext::Batch {
1082                    batch_id,
1083                    cids,
1084                    ready_tx,
1085                } => {
1086                    let broadcast_to = match result {
1087                        Ok(peers) => peers,
1088                        Err(err) => {
1089                            let err: BitswapGetError = err.into();
1090                            log!(
1091                                &task.platform,
1092                                Trace,
1093                                &task.log_target,
1094                                "have broadcast failed (batch), rejecting subscription",
1095                                batch_id = batch_id.0,
1096                                ?err
1097                            );
1098                            task.batches.remove(&batch_id);
1099                            let _ = ready_tx.send(Err(err));
1100                            continue;
1101                        }
1102                    };
1103
1104                    log!(
1105                        &task.platform,
1106                        Trace,
1107                        &task.log_target,
1108                        "have broadcast sent (batch)",
1109                        batch_id = batch_id.0,
1110                        cid_count = cids.len(),
1111                        peers = broadcast_to.len()
1112                    );
1113
1114                    // Under the new ready-handshake protocol, the batch cannot have been
1115                    // cancelled before this point: cancel requires the caller's
1116                    // `BatchCancelGuard`, which is only constructed after we send `Ok(batch_id)`
1117                    // below.
1118                    debug_assert!(task.batches.contains_key(&batch_id));
1119                    if let Some(batch) = task.batches.get_mut(&batch_id) {
1120                        batch.peers_for_cancel = broadcast_to.clone();
1121                    } else {
1122                        // Drop `ready_tx` without sending — the caller's `ready_rx.await.unwrap()`
1123                        // will surface the broken invariant.
1124                        continue;
1125                    }
1126
1127                    // Register one Request per valid slot, sharing the same Have peer set.
1128                    // SipHasher seed drawn once per batch — each slot's HashSet uses the same
1129                    // randomness. Reseeding per slot would only add cost: the seed is for
1130                    // intra-set collision distribution, not cross-set isolation, and the
1131                    // contents (broadcast peer IDs) aren't secrets.
1132                    let timeout = task.platform.now() + Duration::from_secs(10);
1133                    let have_peers_seed = {
1134                        let mut seed = [0; 16];
1135                        task.randomness.fill_bytes(&mut seed);
1136                        seed
1137                    };
1138                    for (slot_idx, cid) in cids {
1139                        let request_id = task.allocate_request_id();
1140                        let have_peers = {
1141                            let mut have_peers = hashbrown::HashSet::with_capacity_and_hasher(
1142                                broadcast_to.len(),
1143                                util::SipHasherBuild::new(have_peers_seed),
1144                            );
1145                            have_peers.extend(broadcast_to.iter().cloned());
1146                            have_peers
1147                        };
1148
1149                        task.requests.insert(
1150                            request_id,
1151                            Request {
1152                                result_tx: SlotOutput::Batch { batch_id, slot_idx },
1153                                timeout: timeout.clone(),
1154                                stage: RequestStage::Have(have_peers),
1155                                cid: cid.clone(),
1156                            },
1157                        );
1158                        task.requests_by_timeout
1159                            .insert((timeout.clone(), request_id));
1160                        task.requests_by_cid
1161                            .entry(cid)
1162                            .or_default()
1163                            .push_back(request_id);
1164
1165                        if let Some(batch) = task.batches.get_mut(&batch_id) {
1166                            batch.slots[slot_idx] = Some(request_id);
1167                        }
1168                    }
1169
1170                    // Signal broadcast success only after per-slot Requests are registered, so
1171                    // events can resolve immediately once the caller installs its handle.
1172                    let _ = ready_tx.send(Ok(batch_id));
1173                }
1174            },
1175            WakeUpReason::NetworkEvent(BitswapEvent::BitswapMessage { peer_id, message }) => {
1176                let message = message.decode();
1177
1178                // Slots that have just resolved and need to be delivered after the per-block-
1179                // presence borrow on `task.requests_by_cid` is released.
1180                let mut deliveries: Vec<(SlotOutput, Result<Vec<u8>, BitswapGetError>)> =
1181                    Vec::new();
1182
1183                for BlockPresence { cid, presence_type } in message.block_presences {
1184                    let cid = match Cid::from_bytes(cid.to_owned()) {
1185                        Ok(cid) => cid,
1186                        Err(error) => {
1187                            log!(
1188                                &task.platform,
1189                                Debug,
1190                                &task.log_target,
1191                                "error decoding CID",
1192                                peer_id,
1193                                error,
1194                            );
1195                            // TODO: Discard entire message? Ban peer? On what errors?
1196                            continue;
1197                        }
1198                    };
1199
1200                    let hashbrown::hash_map::Entry::Occupied(mut entry) =
1201                        task.requests_by_cid.entry(cid.clone())
1202                    else {
1203                        log!(
1204                            &task.platform,
1205                            Trace,
1206                            &task.log_target,
1207                            "stale/unsolicited have response",
1208                            peer_id
1209                        );
1210                        continue;
1211                    };
1212
1213                    log!(
1214                        &task.platform,
1215                        Trace,
1216                        &task.log_target,
1217                        "presence response",
1218                        peer_id,
1219                        cid,
1220                        presence = match presence_type {
1221                            BlockPresenceType::Have => "have",
1222                            BlockPresenceType::DontHave => "dont_have",
1223                        }
1224                    );
1225
1226                    let mut needs_block_request = false;
1227                    let request_ids = entry.get_mut();
1228
1229                    for i in (0..request_ids.len()).rev() {
1230                        let request_id = request_ids[i];
1231                        let request = task.requests.get_mut(&request_id).unwrap();
1232
1233                        match (&mut request.stage, presence_type) {
1234                            (RequestStage::Have(peers), BlockPresenceType::Have) => {
1235                                if peers.contains(&peer_id) {
1236                                    request.stage = RequestStage::Block;
1237                                    needs_block_request = true;
1238                                }
1239                            }
1240                            (RequestStage::Have(peers), BlockPresenceType::DontHave) => {
1241                                let _ = peers.remove(&peer_id);
1242                                if peers.is_empty() {
1243                                    // All peers responded "don't have", fail request.
1244                                    // Normally we shouldn't have more than one request per CID.
1245                                    request_ids.remove(i);
1246                                    let request = task.requests.remove(&request_id).unwrap();
1247                                    let _was_in = task
1248                                        .requests_by_timeout
1249                                        .remove(&(request.timeout, request_id));
1250                                    debug_assert!(_was_in);
1251
1252                                    log!(
1253                                        &task.platform,
1254                                        Trace,
1255                                        &task.log_target,
1256                                        "all peers dont have, NotFound",
1257                                        cid = request.cid
1258                                    );
1259
1260                                    // Defer delivery: dispatching now would re-borrow `task` while
1261                                    // we still hold `entry` on `task.requests_by_cid`.
1262                                    deliveries
1263                                        .push((request.result_tx, Err(BitswapGetError::NotFound)));
1264                                }
1265                            }
1266                            (RequestStage::Block, _) => {}
1267                        }
1268
1269                        // TODO: if at least one request above is in the `Block` stage
1270                        //       already, does this mean we can skip sending another
1271                        //       "block" request?
1272                    }
1273
1274                    if entry.get().is_empty() {
1275                        entry.remove();
1276                    }
1277
1278                    if needs_block_request {
1279                        log!(
1280                            &task.platform,
1281                            Trace,
1282                            &task.log_target,
1283                            "sending block request",
1284                            peer_id,
1285                            cid
1286                        );
1287
1288                        let message = bitswap_block_message(&cid);
1289                        let network_service = task.network_service.clone();
1290                        let peer_id = peer_id.clone();
1291
1292                        task.pending_block_requests.push(Box::pin(async move {
1293                            let result =
1294                                network_service.send_bitswap_message(peer_id, message).await;
1295                            (result, cid)
1296                        }));
1297                    }
1298                }
1299
1300                for Block { prefix, data } in message.payload {
1301                    let prefix = match CidPrefix::from_bytes(prefix.to_owned()) {
1302                        Ok(prefix) => prefix,
1303                        Err(error) => {
1304                            log!(
1305                                &task.platform,
1306                                Debug,
1307                                &task.log_target,
1308                                "error decoding CID prefix",
1309                                peer_id,
1310                                error,
1311                            );
1312                            // TODO: ban peer? On what errors?
1313                            continue;
1314                        }
1315                    };
1316
1317                    let cid = prefix.with_digest_of(data);
1318
1319                    log!(
1320                        &task.platform,
1321                        Trace,
1322                        &task.log_target,
1323                        "block payload received",
1324                        peer_id,
1325                        cid,
1326                        bytes = data.len()
1327                    );
1328
1329                    // Respond to requests asking for this CID regardless of the request stage and
1330                    // remove these requests from internal structures.
1331                    if let Some(request_ids) = task.requests_by_cid.remove(&cid) {
1332                        for request_id in request_ids {
1333                            let request = task.requests.remove(&request_id).unwrap();
1334                            let _was_in = task
1335                                .requests_by_timeout
1336                                .remove(&(request.timeout, request_id));
1337                            debug_assert!(_was_in);
1338
1339                            task.deliver_slot(request.result_tx, Ok(data.to_owned()));
1340                        }
1341                    }
1342                }
1343
1344                // Dispatch deferred deliveries from the block_presences loop.
1345                for (slot_output, result) in deliveries {
1346                    task.deliver_slot(slot_output, result);
1347                }
1348            }
1349            WakeUpReason::BlockRequestResult((result, cid)) => {
1350                // We either succeeded or failed in sending the "block" request.
1351                // Nothing to do on success, but we must respond to requests & cleanup on failure.
1352                if let Err(err) = result {
1353                    log!(
1354                        &task.platform,
1355                        Trace,
1356                        &task.log_target,
1357                        "block request failed",
1358                        cid,
1359                        ?err
1360                    );
1361
1362                    // Requests might have timed out while we were waiting for a response from
1363                    // network service.
1364                    if let Some(request_ids) = task.requests_by_cid.remove(&cid) {
1365                        let err = match err {
1366                            SendBitswapMessageError::QueueFull => BitswapGetError::QueueFull,
1367                            SendBitswapMessageError::NoConnection => {
1368                                BitswapGetError::BlockRequestFailed
1369                            }
1370                        };
1371
1372                        for request_id in request_ids {
1373                            let request = task.requests.remove(&request_id).unwrap();
1374                            let _was_in = task
1375                                .requests_by_timeout
1376                                .remove(&(request.timeout, request_id));
1377                            debug_assert!(_was_in);
1378
1379                            task.deliver_slot(request.result_tx, Err(err.clone()));
1380                        }
1381                    }
1382                }
1383            }
1384            WakeUpReason::RequestTimeout => {
1385                let now = task.platform.now();
1386
1387                let requests = task
1388                    .requests_by_timeout
1389                    .range(..=(now, RequestId::MAX))
1390                    .cloned()
1391                    .collect::<Vec<_>>();
1392
1393                for (timeout, request_id) in requests {
1394                    task.requests_by_timeout.remove(&(timeout, request_id));
1395
1396                    let request = task.requests.remove(&request_id).unwrap();
1397                    let cid = request.cid.clone();
1398
1399                    match task.requests_by_cid.entry(request.cid) {
1400                        hashbrown::hash_map::Entry::Occupied(mut entry) => {
1401                            // The next request to timeout should be always at the front of the
1402                            // queue, but in order to be resistant to platform bugs where `now` is
1403                            // not monothonic (and requests are ordered incorrectly), we use find &
1404                            // remove. It should be almost as fast as `pop_front` if the element is
1405                            // indeed at the front.
1406                            let (index, _) = entry
1407                                .get()
1408                                .iter()
1409                                .find_position(|id| **id == request_id)
1410                                .unwrap();
1411                            entry.get_mut().remove(index);
1412
1413                            if entry.get().is_empty() {
1414                                entry.remove();
1415                            }
1416                        }
1417                        hashbrown::hash_map::Entry::Vacant(_) => unreachable!(),
1418                    }
1419
1420                    log!(
1421                        &task.platform,
1422                        Trace,
1423                        &task.log_target,
1424                        "request timeout",
1425                        cid
1426                    );
1427
1428                    task.deliver_slot(request.result_tx, Err(BitswapGetError::Timeout));
1429                }
1430            }
1431            WakeUpReason::ForegroundClosed => {
1432                // Foreground closed the control channel, end the task.
1433                return;
1434            }
1435        }
1436    }
1437}
