18//! Encoding and decoding of statements for the Statement Store protocol.
19//!
20//! Statements are encoded as `Vec<Field>` where each field has a discriminant byte
21//! followed by field-specific data. Fields must appear in ascending order by discriminant.
22
23use alloc::vec::Vec;
24
25/// Maximum number of topics per statement.
26pub const MAX_TOPICS: usize = 4;
27
28/// `MatchAny` allows to provide a list of topics match against. This is the maximum number of
29/// topics allowed.
30pub const MAX_ANY_TOPICS: usize = 128;
31
32/// Maximum number of statements allowed in a single notification.
33const MAX_STATEMENTS_PER_NOTIFICATION: usize = 10_000;
34
35/// Maximum size in bytes of a single SCALE-encoded statement.
36///
37/// Matches the limit polkadot-sdk enforces.
38pub const MAX_STATEMENT_SIZE: usize = 1024 * 1024 - 1;

40const FIELD_PROOF: u8 = 0;
41const FIELD_DECRYPTION_KEY: u8 = 1;
42const FIELD_EXPIRY: u8 = 2;
43const FIELD_CHANNEL: u8 = 3;
44const FIELD_TOPIC_START: u8 = 4;
45const FIELD_TOPIC_END: u8 = FIELD_TOPIC_START + MAX_TOPICS as u8 - 1;
46const FIELD_DATA: u8 = 8;
47
48const PROOF_SR25519: u8 = 0;
49const PROOF_ED25519: u8 = 1;
50const PROOF_SECP256K1_ECDSA: u8 = 2;
51const PROOF_ON_CHAIN: u8 = 3;

55#[derive(Debug, Clone)]
56pub enum StatementMessage {
57    Statements(Vec<([u8; 32], Statement)>),
58    /// Light nodes send affinity filters to full nodes but don't expect to receive them.
59    ExplicitTopicAffinity(AffinityFilter),
60}

62/// Statement topic (32 bytes).
63pub type Topic = [u8; 32];
64
65/// Decryption key identifier (32 bytes).
66pub type DecryptionKey = [u8; 32];
67
68/// Channel identifier (32 bytes).
69pub type Channel = [u8; 32];
70
71/// Account identifier (32 bytes).
72pub type AccountId = [u8; 32];

74/// Block hash (32 bytes).
75pub type BlockHash = [u8; 32];

77/// A decoded statement.
78#[derive(Debug, Clone, PartialEq, Eq)]
79pub struct Statement {
80    /// Authentication proof for the statement.
81    pub proof: Option<Proof>,
82    /// Identifier for the key that the data field may be decrypted with.
83    pub decryption_key: Option<DecryptionKey>,
84    /// Statement expiry/priority.
85    ///
86    /// The most significant 32 bits represents the expiration timestamp (in seconds since
87    /// UNIX epoch) after which the statement gets removed.
88    /// The lower 32 bits represents an arbitrary sequence number used to order statements
89    /// with the same expiration time.
90    ///
91    /// Higher values indicate a higher priority.
92    pub expiry: u64,
93    /// Account channel. Only one message per (account, channel) pair is allowed.
94    pub channel: Option<Channel>,
95    /// Statement topics (0 to 4).
96    pub topics: Vec<Topic>,
97    /// Additional data.
98    pub data: Option<Vec<u8>>,
99}

101/// Statement proof variants.
102#[derive(Debug, Clone, PartialEq, Eq)]
103pub enum Proof {
104    /// Sr25519 signature proof.
105    Sr25519 {
106        /// The signature (64 bytes).
107        signature: [u8; 64],
108        /// The signer's public key (32 bytes).
109        signer: [u8; 32],
110    },
111    /// Ed25519 signature proof.
112    Ed25519 {
113        /// The signature (64 bytes).
114        signature: [u8; 64],
115        /// The signer's public key (32 bytes).
116        signer: [u8; 32],
117    },
118    /// Secp256k1 ECDSA signature proof.
119    Secp256k1Ecdsa {
120        /// The signature (65 bytes).
121        signature: [u8; 65],
122        /// The signer's public key (33 bytes).
123        signer: [u8; 33],
124    },
125    /// On-chain event proof.
126    OnChain {
127        /// Account identifier associated with the event.
128        who: AccountId,
129        /// Hash of block that contains the event.
130        block_hash: BlockHash,
131        /// Index of the event in the event list.
132        event_index: u64,
133    },
134}

136pub fn statement_hash(statement_bytes: &[u8]) -> [u8; 32] {
137    <[u8; 32]>::try_from(blake2_rfc::blake2b::blake2b(32, &[], statement_bytes).as_bytes())
138        .expect("blake2b output is 32 bytes; qed")
139}

141/// Decodes a statement.
142pub fn decode_statement(bytes: &[u8]) -> Result<Statement, DecodeStatementNotificationError> {
143    match nom::Parser::parse(
144        &mut nom::combinator::all_consuming::<_, nom::error::Error<&[u8]>, _>(
145            nom::combinator::complete(statement_parser),
146        ),
147        bytes,
148    ) {
149        Ok((_, s)) => Ok(s),
150        Err(nom::Err::Error(e) | nom::Err::Failure(e)) => {
151            Err(DecodeStatementNotificationError(e.code))
152        }
153        Err(nom::Err::Incomplete(_)) => {
154            Err(DecodeStatementNotificationError(nom::error::ErrorKind::Eof))
155        }
156    }
157}

159pub fn decode_statement_notification(
160    scale_encoded: &[u8],
161) -> Result<Vec<([u8; 32], Statement)>, DecodeStatementNotificationError> {
162    let (mut remaining, count) = crate::util::nom_scale_compact_usize(scale_encoded).map_err(
163        |_: nom::Err<nom::error::Error<&[u8]>>| {
164            DecodeStatementNotificationError(nom::error::ErrorKind::Fail)
165        },
166    )?;
167
168    if count > MAX_STATEMENTS_PER_NOTIFICATION {
169        return Err(DecodeStatementNotificationError(
170            nom::error::ErrorKind::TooLarge,
171        ));
172    }
173
174    let mut statements = Vec::with_capacity(count);
175    for _ in 0..count {
176        let start = remaining;
177        let (rest, statement) = statement_parser(remaining).map_err(|e| match e {
178            nom::Err::Error(e) | nom::Err::Failure(e) => DecodeStatementNotificationError(e.code),
179            nom::Err::Incomplete(_) => DecodeStatementNotificationError(nom::error::ErrorKind::Eof),
180        })?;
181        let raw = &start[..start.len() - rest.len()];
182        let hash = statement_hash(raw);
183        statements.push((hash, statement));
184        remaining = rest;
185    }
186
187    if !remaining.is_empty() {
188        return Err(DecodeStatementNotificationError(
189            nom::error::ErrorKind::NonEmpty,
190        ));
191    }
192
193    Ok(statements)
194}

214const V2_TAG_STATEMENTS: u8 = 0x00;
215const V2_TAG_AFFINITY: u8 = 0x01;
216
217pub fn decode_statement_message(
218    bytes: &[u8],
219) -> Result<StatementMessage, DecodeStatementMessageError> {
220    if bytes.is_empty() {
221        return Err(DecodeStatementMessageError::Empty);
222    }
223
224    match bytes[0] {
225        V2_TAG_STATEMENTS => {
226            let stmts = decode_statement_notification(&bytes[1..])
227                .map_err(DecodeStatementMessageError::InvalidStatements)?;
228            Ok(StatementMessage::Statements(stmts))
229        }
230        V2_TAG_AFFINITY => {
231            let filter = AffinityFilter::decode(&bytes[1..])
232                .map_err(|_| DecodeStatementMessageError::InvalidBloomFilter)?;
233            Ok(StatementMessage::ExplicitTopicAffinity(filter))
234        }
235        other => Err(DecodeStatementMessageError::UnknownVariant(other)),
236    }
237}

239pub fn encode_statements_message(statements: &[&[u8]]) -> Vec<u8> {
240    let tag_len = 1;
241    let max_compact_len = 5;
242    let total_len: usize = statements.iter().map(|s| s.len()).sum();
243    let mut out = Vec::with_capacity(tag_len + max_compact_len + total_len);
244    out.push(V2_TAG_STATEMENTS);
245    out.extend_from_slice(crate::util::encode_scale_compact_usize(statements.len()).as_ref());
246    for stmt in statements {
247        out.extend_from_slice(stmt);
248    }
249    out
250}

252pub fn encode_topic_affinity_message(filter: &AffinityFilter) -> Vec<u8> {
253    let tag_len = 1;
254    let encoded = filter.encode_to_vec();
255    let mut out = Vec::with_capacity(tag_len + encoded.len());
256    out.push(V2_TAG_AFFINITY);
257    out.extend_from_slice(&encoded);
258    out
259}

273/// Encodes a single statement.
274pub fn encode_statement(statement: &Statement) -> Result<Vec<u8>, EncodeStatementError> {
275    let mut out = Vec::new();
276    encode_statement_into(statement, &mut out)?;
277    Ok(out)
278}

280fn encode_statement_into(
281    statement: &Statement,
282    out: &mut Vec<u8>,
283) -> Result<(), EncodeStatementError> {
284    if statement.topics.len() > MAX_TOPICS {
285        return Err(EncodeStatementError::TooManyTopics {
286            got: statement.topics.len(),
287            max: MAX_TOPICS,
288        });
289    }
290
291    let num_fields = statement.proof.is_some() as usize
292        + statement.decryption_key.is_some() as usize
293        + 1 // expiry is always present
294        + statement.channel.is_some() as usize
295        + statement.topics.len()
296        + statement.data.is_some() as usize;
297
298    out.extend_from_slice(crate::util::encode_scale_compact_usize(num_fields).as_ref());
299
300    if let Some(proof) = &statement.proof {
301        out.push(FIELD_PROOF);
302        encode_proof_into(proof, out);
303    }
304
305    if let Some(key) = &statement.decryption_key {
306        out.push(FIELD_DECRYPTION_KEY);
307        out.extend_from_slice(key);
308    }
309
310    out.push(FIELD_EXPIRY);
311    out.extend_from_slice(&statement.expiry.to_le_bytes());
312
313    if let Some(channel) = &statement.channel {
314        out.push(FIELD_CHANNEL);
315        out.extend_from_slice(channel);
316    }
317
318    for (i, topic) in statement.topics.iter().enumerate() {
319        out.push(FIELD_TOPIC_START + i as u8);
320        out.extend_from_slice(topic);
321    }
322
323    if let Some(data) = &statement.data {
324        out.push(FIELD_DATA);
325        out.extend_from_slice(crate::util::encode_scale_compact_usize(data.len()).as_ref());
326        out.extend_from_slice(data);
327    }
328
329    Ok(())
330}
331
332fn encode_proof_into(proof: &Proof, out: &mut Vec<u8>) {
333    match proof {
334        Proof::Sr25519 { signature, signer } => {
335            out.push(PROOF_SR25519);
336            out.extend_from_slice(signature.as_slice());
337            out.extend_from_slice(signer.as_slice());
338        }
339        Proof::Ed25519 { signature, signer } => {
340            out.push(PROOF_ED25519);
341            out.extend_from_slice(signature.as_slice());
342            out.extend_from_slice(signer.as_slice());
343        }
344        Proof::Secp256k1Ecdsa { signature, signer } => {
345            out.push(PROOF_SECP256K1_ECDSA);
346            out.extend_from_slice(signature.as_slice());
347            out.extend_from_slice(signer.as_slice());
348        }
349        Proof::OnChain {
350            who,
351            block_hash,
352            event_index,
353        } => {
354            out.push(PROOF_ON_CHAIN);
355            out.extend_from_slice(who.as_slice());
356            out.extend_from_slice(block_hash.as_slice());
357            out.extend_from_slice(&event_index.to_le_bytes());
358        }
359    }
360}

364fn statement_parser(input: &[u8]) -> nom::IResult<&[u8], Statement> {
365    let (input, num_fields) = crate::util::nom_scale_compact_usize(input)?;
366    fields_parser(num_fields)(input)
367}
368
369fn fields_parser(num_fields: usize) -> impl FnMut(&[u8]) -> nom::IResult<&[u8], Statement> {
370    move |mut input: &[u8]| {
371        let mut proof = None;
372        let mut decryption_key = None;
373        let mut expiry = None;
374        let mut channel = None;
375        let mut topics = Vec::new();
376        let mut data = None;
377
378        let mut last_tag: Option<u8> = None;
379
380        for _ in 0..num_fields {
381            let (rest, tag) = nom::number::streaming::u8(input)?;
382
383            if let Some(lt) = last_tag {
384                if tag <= lt {
385                    return Err(nom::Err::Failure(nom::error::make_error(
386                        input,
387                        nom::error::ErrorKind::Verify,
388                    )));
389                }
390            }
391            last_tag = Some(tag);
392
393            let rest = match tag {
394                FIELD_PROOF => {
395                    let (rest, p) = proof_parser(rest)?;
396                    proof = Some(p);
397                    rest
398                }
399                FIELD_DECRYPTION_KEY => {
400                    let (rest, key) = nom::bytes::streaming::take(32u32)(rest)?;
401                    decryption_key =
402                        Some(<[u8; 32]>::try_from(key).expect("take(32) guarantees 32 bytes; qed"));
403                    rest
404                }
405                FIELD_EXPIRY => {
406                    let (rest, exp) = nom::number::streaming::le_u64(rest)?;
407                    expiry = Some(exp);
408                    rest
409                }
410                FIELD_CHANNEL => {
411                    let (rest, ch) = nom::bytes::streaming::take(32u32)(rest)?;
412                    channel =
413                        Some(<[u8; 32]>::try_from(ch).expect("take(32) guarantees 32 bytes; qed"));
414                    rest
415                }
416                FIELD_TOPIC_START..=FIELD_TOPIC_END => {
417                    let topic_index = (tag - FIELD_TOPIC_START) as usize;
418                    if topic_index != topics.len() {
419                        return Err(nom::Err::Failure(nom::error::make_error(
420                            input,
421                            nom::error::ErrorKind::Verify,
422                        )));
423                    }
424                    let (rest, topic) = nom::bytes::streaming::take(32u32)(rest)?;
425                    topics.push(
426                        <[u8; 32]>::try_from(topic).expect("take(32) guarantees 32 bytes; qed"),
427                    );
428                    rest
429                }
430                FIELD_DATA => {
431                    let (rest, len) = crate::util::nom_scale_compact_usize(rest)?;
432                    let (rest, d) = nom::bytes::streaming::take(len)(rest)?;
433                    data = Some(d.to_vec());
434                    rest
435                }
436                _ => {
437                    return Err(nom::Err::Failure(nom::error::make_error(
438                        input,
439                        nom::error::ErrorKind::Verify,
440                    )));
441                }
442            };
443
444            input = rest;
445        }
446
447        let expiry = expiry.ok_or_else(|| {
448            nom::Err::Failure(nom::error::make_error(input, nom::error::ErrorKind::Verify))
449        })?;
450
451        let statement = Statement {
452            proof,
453            decryption_key,
454            expiry,
455            channel,
456            topics,
457            data,
458        };
459
460        Ok((input, statement))
461    }
462}

464fn proof_parser(input: &[u8]) -> nom::IResult<&[u8], Proof> {
465    let (input, variant) = nom::number::streaming::u8(input)?;
466    match variant {
467        PROOF_SR25519 => {
468            let (input, signature) = nom::bytes::streaming::take(64u32)(input)?;
469            let (input, signer) = nom::bytes::streaming::take(32u32)(input)?;
470            Ok((
471                input,
472                Proof::Sr25519 {
473                    signature: <[u8; 64]>::try_from(signature)
474                        .expect("take(64) guarantees 64 bytes; qed"),
475                    signer: <[u8; 32]>::try_from(signer)
476                        .expect("take(32) guarantees 32 bytes; qed"),
477                },
478            ))
479        }
480        PROOF_ED25519 => {
481            let (input, signature) = nom::bytes::streaming::take(64u32)(input)?;
482            let (input, signer) = nom::bytes::streaming::take(32u32)(input)?;
483            Ok((
484                input,
485                Proof::Ed25519 {
486                    signature: <[u8; 64]>::try_from(signature)
487                        .expect("take(64) guarantees 64 bytes; qed"),
488                    signer: <[u8; 32]>::try_from(signer)
489                        .expect("take(32) guarantees 32 bytes; qed"),
490                },
491            ))
492        }
493        PROOF_SECP256K1_ECDSA => {
494            let (input, signature) = nom::bytes::streaming::take(65u32)(input)?;
495            let (input, signer) = nom::bytes::streaming::take(33u32)(input)?;
496            Ok((
497                input,
498                Proof::Secp256k1Ecdsa {
499                    signature: <[u8; 65]>::try_from(signature)
500                        .expect("take(65) guarantees 65 bytes; qed"),
501                    signer: <[u8; 33]>::try_from(signer)
502                        .expect("take(33) guarantees 33 bytes; qed"),
503                },
504            ))
505        }
506        PROOF_ON_CHAIN => {
507            let (input, who) = nom::bytes::streaming::take(32u32)(input)?;
508            let (input, block_hash) = nom::bytes::streaming::take(32u32)(input)?;
509            let (input, event_index) = nom::number::streaming::le_u64(input)?;
510            Ok((
511                input,
512                Proof::OnChain {
513                    who: <[u8; 32]>::try_from(who).expect("take(32) guarantees 32 bytes; qed"),
514                    block_hash: <[u8; 32]>::try_from(block_hash)
515                        .expect("take(32) guarantees 32 bytes; qed"),
516                    event_index,
517                },
518            ))
519        }
520        _ => Err(nom::Err::Failure(nom::error::make_error(
521            input,
522            nom::error::ErrorKind::Verify,
523        ))),
524    }
525}

806             0101010101010101010101010101010101010101010101010101010101010105020202020202\
807             020202020202020202020202020202020202020202020202020208083763",
808        )
809        .unwrap();
810
811        let (remaining, decoded) = statement_parser(&bytes).unwrap();
812        assert!(remaining.is_empty());
813
814        assert!(matches!(
815            decoded.proof,
816            Some(Proof::OnChain { who, block_hash, event_index })
817            if who == [42u8; 32]
818                && block_hash == [24u8; 32]
819                && event_index == 66
820        ));
821        assert_eq!(decoded.decryption_key, Some([0xde; 32]));
822        assert_eq!(decoded.topics.len(), 2);
823        assert_eq!(decoded.topics[0], [0x01; 32]);
824        assert_eq!(decoded.topics[1], [0x02; 32]);
825        assert_eq!(decoded.data.as_deref(), Some([55, 99].as_slice()));
826        assert_eq!(decoded.expiry, 999);
827        assert_eq!(decoded.channel, Some([0xcc; 32]));
828

830
831        // Ed25519 proof: signature=[0xAA; 64], signer=[0xBB; 32], expiry=1000, data=[1,2,3]

833        let ed25519_bytes = hex::decode(
834            "0c0001aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa\
835            aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaabbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb\
836            bbbbbbbbbbbbbbbbbbbbbbbbbbb02e803000000000000080c010203",
837        )

841        assert!(remaining.is_empty());
