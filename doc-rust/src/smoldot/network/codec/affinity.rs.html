20use core::hash::{BuildHasher, Hasher};

23/// Maximum number of bits allowed in a bloom filter received from the network.
24/// 1 MiB (the notification size budget) = 8_388_608 bits.
25const MAX_BLOOM_BITS: usize = 1024 * 1024 * 8;
26
27/// Maximum number of hash functions allowed.
28/// Optimal hash count is `(bits / items) * ln(2)`. With the minimum allocation of 64 bits
29/// and 1 expected item this yields ~44, so the limit must be at least that high. 64 covers all
30/// practical configurations while preventing CPU abuse from peers.
31pub(crate) const MAX_NUM_HASHES: u32 = 64;
32
33/// Minimum number of expected items used to size a filter built from a local topic list.
34///
35/// Yields ~160 bytes at FPR=0.01.
36const MIN_EXPECTED_ITEMS: usize = 128;
37
38/// Wraps `fastbloom::DefaultHasher` to force `write_usize`/`write_isize` to always emit
39/// 8-byte LE values, ensuring identical bloom filter bits on wasm32 and 64-bit native targets.
40#[derive(Clone, Debug)]
41struct PortableBuildHasher(BloomDefaultHasher);
42
43impl PortableBuildHasher {
44    fn seeded(seed: u128) -> Self {
45        Self(BloomDefaultHasher::seeded(&seed.to_le_bytes()))
46    }
47}
48
49impl BuildHasher for PortableBuildHasher {
50    type Hasher = PortableHasher;
51
52    fn build_hasher(&self) -> Self::Hasher {
53        PortableHasher(self.0.build_hasher())
54    }
55}
56
57/// Hasher state returned by [`PortableBuildHasher`].  Delegates everything to
58/// the inner SipHash-based hasher but overrides `write_usize` and `write_isize`
59/// so that platform-width integers are always 8 bytes regardless of pointer
60/// width.
61#[derive(Clone)]
62struct PortableHasher(<BloomDefaultHasher as BuildHasher>::Hasher);
63
64impl Hasher for PortableHasher {
65    #[inline]
66    fn finish(&self) -> u64 {
67        self.0.finish()
68    }
69
70    #[inline]
71    fn write(&mut self, bytes: &[u8]) {
72        self.0.write(bytes);
73    }
74
75    #[inline]
76    fn write_usize(&mut self, i: usize) {
77        // Always write as 8-byte little-endian so that `wasm32` (4-byte
78        // usize) and 64-bit targets produce the same hash.
79        self.0.write(&(i as u64).to_le_bytes());
80    }
81
82    #[inline]
83    fn write_isize(&mut self, i: isize) {
84        // Always write as 8-byte little-endian for the same reason as
85        // `write_usize`.
86        self.0.write(&(i as i64).to_le_bytes());
87    }
88}

90/// Wire representation of a bloom filter.
91struct EncodedBloomFilter {
92    // Seed used for hashing items in the bloom filter. Needed for the peer to reconstruct the same
93    // bloom filter.
94    seed: u128,
95    // Number of hash functions used in the bloom filter. Needed for the peer to reconstruct the
96    // same bloom filter.
97    num_hashes: u32,
98    // Bloom filter bits as a vector of u64. The bloom filter is reconstructed by the peer using
99    // these bits.
100    bits: Vec<u64>,
101}
102
103impl EncodedBloomFilter {
104    fn encode_to_vec(&self) -> Vec<u8> {
105        let mut out = Vec::new();
106        out.extend_from_slice(&self.seed.to_le_bytes());
107        out.extend_from_slice(&self.num_hashes.to_le_bytes());
108        out.extend_from_slice(crate::util::encode_scale_compact_usize(self.bits.len()).as_ref());
109        for &word in &self.bits {
110            out.extend_from_slice(&word.to_le_bytes());
111        }
112        out
113    }
114
115    fn decode(data: &[u8]) -> Result<Self, DecodeAffinityFilterError> {
116        if data.len() < 20 {
117            return Err(DecodeAffinityFilterError);
118        }
119        let seed = u128::from_le_bytes(<[u8; 16]>::try_from(&data[..16]).unwrap());
120        let num_hashes = u32::from_le_bytes(<[u8; 4]>::try_from(&data[16..20]).unwrap());
121        let rest = &data[20..];
122        let (rest, bits_len) =
123            crate::util::nom_scale_compact_usize::<nom::error::Error<&[u8]>>(rest)
124                .map_err(|_| DecodeAffinityFilterError)?;
125        if rest.len() != bits_len * 8 {
126            return Err(DecodeAffinityFilterError);
127        }
128        let mut bits = Vec::with_capacity(bits_len);
129        for chunk in rest.chunks_exact(8) {
130            bits.push(u64::from_le_bytes(<[u8; 8]>::try_from(chunk).unwrap()));
131        }
132        Ok(EncodedBloomFilter {
133            seed,
134            num_hashes,
135            bits,
136        })
137    }
138}

140#[derive(Debug, Clone)]
141pub struct AffinityFilter {
142    /// Bloom filter bytes representing the topics this peer is interested in.
143    bloom: fastbloom::BloomFilter<PortableBuildHasher>,
144    /// Seed used for hashing items in the bloom filter.
145    seed: u128,
146}
147
148impl AffinityFilter {
149    pub fn new(seed: u128, false_pos: f64, expected_items: usize) -> Self {
150        let bloom = fastbloom::BloomFilter::with_false_pos(false_pos)
151            .hasher(PortableBuildHasher::seeded(seed))
152            .expected_items(expected_items);
153        AffinityFilter { bloom, seed }
154    }
155
156    pub fn from_topics<'a>(
157        topics: impl Iterator<Item = &'a [u8; 32]>,
158        seed: u128,
159        false_positive_rate: f64,
160    ) -> Self {
161        let topics: Vec<&[u8; 32]> = topics.collect();
162        let count = topics.len().max(MIN_EXPECTED_ITEMS);
163        let mut filter = Self::new(seed, false_positive_rate, count);
164        for topic in topics {
165            filter.insert(topic);
166        }
167        filter
168    }
169
170    pub fn decode(data: &[u8]) -> Result<Self, DecodeAffinityFilterError> {
171        let encoded = EncodedBloomFilter::decode(data)?;
172        if encoded.bits.is_empty() {
173            return Err(DecodeAffinityFilterError);
174        }
175        if encoded.bits.len() * u64::BITS as usize > MAX_BLOOM_BITS {
176            return Err(DecodeAffinityFilterError);
177        }
178        if encoded.num_hashes == 0 || encoded.num_hashes > MAX_NUM_HASHES {
179            return Err(DecodeAffinityFilterError);
180        }
181        let bloom = fastbloom::BloomFilter::from_vec(encoded.bits)
182            .hasher(PortableBuildHasher::seeded(encoded.seed))
183            .hashes(encoded.num_hashes);
184        Ok(AffinityFilter {
185            bloom,
186            seed: encoded.seed,
187        })
188    }
189
190    /// Insert a topic into the bloom filter.
191    pub fn insert(&mut self, topic: &[u8; 32]) {
192        self.bloom.insert(topic);
193    }
194
195    /// Check if a topic is likely present in the bloom filter.
196    pub fn contains(&self, topic: &[u8; 32]) -> bool {
197        self.bloom.contains(topic)
198    }
199
200    /// Check if topics match this affinity filter.
201    ///
202    /// Topics match if any of them is present in the bloom filter.
203    /// No topics always match (from broadcast statements).
204    pub fn matches_statement(&self, topics: &[&Topic]) -> bool {
205        if topics.is_empty() {
206            return true;
207        }
208        topics.iter().any(|t| self.bloom.contains(t))
209    }
210
211    pub fn encode_to_vec(&self) -> Vec<u8> {
212        debug_assert!((1..=MAX_NUM_HASHES).contains(&self.bloom.num_hashes()));
213        let encoded = EncodedBloomFilter {
214            seed: self.seed,
215            num_hashes: self.bloom.num_hashes(),
216            bits: self.bloom.as_slice().to_vec(),
217        };
218        encoded.encode_to_vec()
219    }
220
221    pub fn match_all(seed: u128) -> Self {
222        let bits = alloc::vec![u64::MAX; 16];
223        let bloom = fastbloom::BloomFilter::from_vec(bits)
224            .hasher(PortableBuildHasher::seeded(seed))
225            .hashes(1);
226        AffinityFilter { bloom, seed }
227    }
228}

310    #[test]
311    fn large_roundtrip() {
312        const TOTAL: usize = 100_000;
313        const SET_COUNT: usize = TOTAL / 10;
314
315        let items: Vec<[u8; 32]> = (0..TOTAL)
316            .map(|i| {
317                let mut key = [0u8; 32];
318                key[..8].copy_from_slice(&(i as u64).to_le_bytes());
319                key
320            })
321            .collect();
322
323        let mut filter = AffinityFilter::new(TEST_SEED, BLOOM_FALSE_POS_RATE, SET_COUNT);
324        for item in &items[..SET_COUNT] {
325            filter.insert(item);
326        }
327
328        let expected: Vec<bool> = items.iter().map(|item| filter.contains(item)).collect();
329        for i in 0..SET_COUNT {
330            assert!(expected[i], "inserted item {i} must be present");
331        }
332
333        let encoded = filter.encode_to_vec();
334        let decoded = AffinityFilter::decode(&encoded).expect("decoding should succeed");
335
336        for (i, item) in items.iter().enumerate() {
337            assert_eq!(decoded.contains(item), expected[i], "mismatch for item {i}");
338        }
339
340        assert_eq!(
341            encoded,
342            decoded.encode_to_vec(),
343            "re-encoding should produce identical bytes"
344        );
345    }

349        const ITEM_COUNT: usize = 10_000;
350
351        let items: Vec<[u8; 32]> = (0..ITEM_COUNT)
352            .map(|i| {

376            "blake2_256 digest must match polkadot-sdk snapshot"

379        let decoded = AffinityFilter::decode(&encoded).expect("snapshot must decode");
380        for (i, item) in items.iter().enumerate() {

385        }

423    #[test]
424    fn decode_rejects_truncated_bits_body() {
425        // The compact length claims 2 u64 words (16 bytes of bits) but only 8 bytes follow;
426        // `EncodedBloomFilter::decode` guards against this via the `rest.len() != bits_len * 8`
427        // check. polkadot-sdk delegates to `codec::Decode` which handles incomplete input
428        // automatically; smoldot parses peer bytes by hand and needs explicit coverage.
429        let mut bytes = Vec::new();
430        bytes.extend_from_slice(&TEST_SEED.to_le_bytes());
431        bytes.extend_from_slice(&1u32.to_le_bytes());
432        bytes.extend_from_slice(crate::util::encode_scale_compact_usize(2).as_ref());
433        bytes.extend_from_slice(&0u64.to_le_bytes());
434        assert!(AffinityFilter::decode(&bytes).is_err());
435    }

437    #[test]
438    fn from_topics_applies_minimum_floor() {
439        let reference = AffinityFilter::new(TEST_SEED, BLOOM_FALSE_POS_RATE, MIN_EXPECTED_ITEMS)
440            .encode_to_vec();
441
442        for topic_count in [0usize, 1, MIN_EXPECTED_ITEMS / 2, MIN_EXPECTED_ITEMS] {
443            let topics: Vec<[u8; 32]> = (0..topic_count)
444                .map(|i| {
445                    let mut key = [0u8; 32];
446                    key[..8].copy_from_slice(&(i as u64).to_le_bytes());
447                    key
448                })
449                .collect();
450            let filter =
451                AffinityFilter::from_topics(topics.iter(), TEST_SEED, BLOOM_FALSE_POS_RATE);
452            assert_eq!(
453                filter.encode_to_vec().len(),
454                reference.len(),
455                "encoded size for {topic_count} topics must match the floor-sized filter",
456            );
457        }
458
459        let above_floor = MIN_EXPECTED_ITEMS * 4;
460        let topics: Vec<[u8; 32]> = (0..above_floor)
461            .map(|i| {
462                let mut key = [0u8; 32];
463                key[..8].copy_from_slice(&(i as u64).to_le_bytes());
464                key
465            })
466            .collect();
467        let filter = AffinityFilter::from_topics(topics.iter(), TEST_SEED, BLOOM_FALSE_POS_RATE);
468        assert!(
469            filter.encode_to_vec().len() > reference.len(),
470            "filter sized above the floor must produce a larger encoding",
471        );
472    }
