18//! Event-fed local view of the statement peers of a chain, the light client's port of the
19//! `peers_topology` module of the full node's statement protocol.
20//!
21//! Every peer is mapped into the 32-byte topic key space by the blake2b-256 hash of its
22//! [`PeerId`], and the distance to a topic is the XOR of the two keys. The full nodes keep a
23//! topic on the `replication_factor` peers closest to it, so a light client that connects to
24//! one of the closest known peers for a topic is likely to reach a holder of that topic.
25//!
26//! The topology is built from peers learned through discovery, Identify and statement
27//! substreams, and remembers them with their addresses after they disconnect, so that a peer
28//! close to a topic can be connected again later, even once the address book shared with the
29//! block sync has dropped it. It computes XOR distances locally over that learned peer set and
30//! never issues a topic-specific lookup, so that nobody learns which topics the local node
31//! subscribes to.
32//!
33//! The full node counts a peer as a statement peer only once Identify confirms it. A light
34//! client connects to a handful of peers and gets no Identify answer from the other ones, so a
35//! discovered peer counts as a statement peer until Identify or a refused statement substream
36//! says otherwise.
37//!
38//! Only a full node on the DHT path speaks `statement/3`. A peer on an older version wants
39//! every statement and is no candidate for a topic, unless no known peer speaks `statement/3`.
40//!
41//! A light client is never a replica of a topic, so the affinity oracle and the routing of
42//! statements between replicas of the full node have no counterpart here.
43

57/// A point in the 32-byte key space shared by topics and hashed peer ids.
58pub type Key = [u8; 32];
59
60/// Evict a disconnected peer unseen by any event for this long.
61const PEER_STALENESS_TTL: Duration = Duration::from_secs(2 * 60 * 60);
62
63/// Hard cap on the known peers. Bounds memory when discovery outruns staleness eviction.
64const MAX_KNOWN_PEERS: usize = 8192;
65
66/// Addresses remembered per peer, the newest ones.
67const MAX_ADDRESSES_PER_PEER: usize = 10;

69/// Configuration passed to [`PeersTopology::new`].
70pub struct Config {
71    /// Seed used for the randomness of the hash map holding the peers.
72    pub randomness_seed: [u8; 32],
73
74    /// Number of statement peers responsible for storing a topic, the replication factor of
75    /// the full nodes.
76    pub replication_factor: NonZeroUsize,
77}

79#[derive(Debug)]
80struct PeerInfo<TNow> {
81    /// `false` once the peer is known to lack the statement protocol.
82    supports_protocol: bool,
83    /// Whether the peer speaks `statement/3`. `None` until Identify or a substream tells.
84    dht_path: Option<bool>,
85    /// The statement substream to the peer is open.
86    connected: bool,
87    /// Cached `peer_key`. The peer id never changes and hashing is costly.
88    key: Key,
89    /// Time of the most recent event observing this peer, the eviction key for both staleness
90    /// and the capacity backstop.
91    last_seen: TNow,
92    /// Multiaddresses the peer was discovered with, at most `MAX_ADDRESSES_PER_PEER`, the
93    /// newest last.
94    addresses: Vec<Vec<u8>>,
95}
96
97/// Local view of the statement peers of a chain. See the [module documentation](self).
98///
99/// Events only add peers. The caller runs [`PeersTopology::evict`] periodically to drop stale
100/// peers and to hold the cap on the known peers.
101#[derive(Debug)]
102pub struct PeersTopology<TNow> {
103    replication_factor: NonZeroUsize,
104    /// Known remote peers, evicted by [`PeersTopology::evict`] once stale or over
105    /// `MAX_KNOWN_PEERS`.
106    discovered: hashbrown::HashMap<PeerId, PeerInfo<TNow>, util::SipHasherBuild>,
107}
108
109impl<TNow> PeersTopology<TNow>
110where
111    TNow: Clone + Ord + Add<Duration, Output = TNow>,
112{
113    /// Builds a new topology that knows no peer.
114    pub fn new(config: Config) -> Self {
115        let mut randomness = ChaCha20Rng::from_seed(config.randomness_seed);
116
117        PeersTopology {
118            replication_factor: config.replication_factor,
119            discovered: hashbrown::HashMap::with_hasher(util::SipHasherBuild::new({
120                let mut seed = [0; 16];
121                randomness.fill_bytes(&mut seed);
122                seed
123            })),
124        }
125    }

127    /// Record that a `FindNode` response listed `peers`, each with its multiaddresses.
128    pub fn on_peers_discovered<TAddrs>(
129        &mut self,
130        peers: impl IntoIterator<Item = (PeerId, TAddrs)>,
131        now: &TNow,
132    ) where
133        TAddrs: IntoIterator<Item = Vec<u8>>,
134    {
135        for (peer, addresses) in peers {
136            let info = self.get_or_insert_peer(peer, now);
137            for address in addresses {
138                // An address reported again moves to the newest end.
139                if let Some(position) = info.addresses.iter().position(|a| *a == address) {
140                    info.addresses.remove(position);
141                } else if info.addresses.len() >= MAX_ADDRESSES_PER_PEER {
142                    info.addresses.remove(0);
143                }
144                info.addresses.push(address);
145            }
146        }
147    }

149    /// Record `statement/3` support from the Identify response, or the lack of the statement
150    /// protocol from a statement substream refused for an unknown protocol.
151    ///
152    /// Peers that do not support the statement protocol remain known but are excluded from
153    /// the candidates for a topic. An open statement substream proves support and outweighs
154    /// either answer.
155    pub fn on_peer_identified(
156        &mut self,
157        peer: PeerId,
158        supports_statement_protocol: bool,
159        now: &TNow,
160    ) {
161        let info = self.get_or_insert_peer(peer, now);
162        if info.connected {
163            return;
164        }
165        info.supports_protocol = supports_statement_protocol;
166        if supports_statement_protocol {
167            info.dht_path = Some(true);
168        }
169    }

171    /// Record that the statement substream to `peer` opened, on `statement/3` if `dht_path`.
172    ///
173    /// An open substream implies statement protocol support.
174    pub fn on_substream_opened(&mut self, peer: PeerId, dht_path: bool, now: &TNow) {
175        let info = self.get_or_insert_peer(peer, now);
176        info.supports_protocol = true;
177        info.dht_path = Some(dht_path);
178        info.connected = true;
179    }

181    /// Record that the statement substream to `peer` closed.
182    ///
183    /// A peer on an older version keeps its mark, or it would be connected again at once.
184    pub fn on_substream_closed(&mut self, peer: PeerId, now: &TNow) {
185        if let Some(info) = self.discovered.get_mut(&peer) {
186            info.connected = false;
187            if *now > info.last_seen {
188                info.last_seen = now.clone();
189            }
190        }
191    }

193    /// Number of known remote peers, including peers without the statement protocol.
194    pub fn known_peers_count(&self) -> usize {
195        self.discovered.len()
196    }

198    /// Known candidates for a topic, in no specific order.

203    /// Multiaddresses `peer` was discovered with. Empty for an unknown peer, or for a peer
204    /// learned only from Identify or from a statement substream.
205    pub fn peer_addresses(&self, peer: &PeerId) -> impl Iterator<Item = &[u8]> {
206        self.discovered
207            .get(peer)
208            .into_iter()
209            .flat_map(|info| info.addresses.iter().map(Vec::as_slice))
210    }

212    /// Closest known statement peers for `topic`.
213    ///
214    /// "Closest" is computed over the locally learned statement peers, not by querying the
215    /// network for the true global closest peers.
216    fn closest_known(&self, topic: &Key, limit: usize) -> Vec<PeerId> {
217        self.closest_known_keyed(topic, limit)
218            .into_iter()
219            .map(|(peer, _)| peer)
220            .collect()
221    }

223    /// Local-only connection candidates for `topics`: a small set of peers, chosen greedily,
224    /// covering every topic, each topic being covered by any of its `replication_factor`
225    /// closest known peers.
226    ///
227    /// Only the locally learned topology is used, avoiding network lookups that would reveal
228    /// the topics. The result does not depend on which peers are connected.
229    pub fn peers_for_topics(&self, topics: &[Key]) -> Vec<PeerId> {
230        let mut uncovered = topics
231            .iter()
232            .map(|topic| {
233                let pool = self.closest_known_keyed(topic, self.replication_factor.get());
234                (topic, pool)
235            })
236            .collect::<Vec<_>>();
237
238        // A selected peer covers every topic whose pool holds it, so it never comes up again and
239        // each round covers at least one more topic.
240        let mut selected = Vec::new();
241        while let Some(best_peer) = best_candidate(&uncovered) {
242            uncovered.retain(|(_, pool)| !pool_contains(pool, &best_peer));
243            selected.push(best_peer);
244        }
245        selected
246    }
247
248    /// Connected peers to send a statement on `topics` to: every connected peer among the
249    /// `replication_factor` closest known statement peers of one of the topics, and every
250    /// connected peer on an older version.
251    ///
252    /// The full node also routes to connected peers closer to a topic than itself. A light
253    /// client is no replica and has no position in the key space, so it keeps only the replicas.
254    pub fn routing_targets(&self, topics: &[Key]) -> BTreeSet<PeerId> {
255        let without_dht = self
256            .discovered
257            .iter()
258            .filter(|(_, info)| info.dht_path == Some(false))
259            .map(|(peer, _)| peer.clone());
260        topics
261            .iter()
262            .flat_map(|topic| self.closest_known(topic, self.replication_factor.get()))
263            .chain(without_dht)
264            .filter(|peer| self.discovered.get(peer).is_some_and(|info| info.connected))
265            .collect()
266    }

268    /// Evict disconnected peers unseen for `PEER_STALENESS_TTL` as of `now`, plus any excess
269    /// over `MAX_KNOWN_PEERS`, least recently seen first.
270    /// Returns whether the candidates for a topic changed.
271    pub fn evict(&mut self, now: &TNow) -> bool {
272        // The candidates change only when a candidate goes, and `dht_path_known` holds until
273        // the first of them does.
274        let dht_path_known = self.dht_path_known();
275        let mut changed = false;
276        self.discovered.retain(|_, info| {
277            let keep = info.connected || info.last_seen.clone() + PEER_STALENESS_TTL > *now;
278            changed |= !keep && info.is_candidate(dht_path_known);
279            keep
280        });
281
282        let excess = self.discovered.len().saturating_sub(MAX_KNOWN_PEERS);
283        if excess == 0 {
284            return changed;
285        }
286        let mut victims = self
287            .discovered
288            .iter()
289            .filter(|(_, info)| !info.connected)
290            .map(|(peer, info)| (info.last_seen.clone(), peer.clone()))
291            .collect::<Vec<_>>();
292        // Connected peers are never evicted, so there may be fewer victims than `excess`.
293        let excess = excess.min(victims.len());
294        if excess > 0 {
295            victims.select_nth_unstable(excess - 1);
296        }
297        for (_, peer) in victims.into_iter().take(excess) {
298            if let Some(info) = self.discovered.remove(&peer) {
299                changed |= info.is_candidate(dht_path_known);
300            }
301        }
302        changed
303    }

305    /// Insert `peer` if absent, refresh its `last_seen`, and return its record.
306    ///
307    /// An event delivered late, with a `now` older than the last one, leaves `last_seen` alone.
308    fn get_or_insert_peer(&mut self, peer: PeerId, now: &TNow) -> &mut PeerInfo<TNow> {
309        let info = self
310            .discovered
311            .entry(peer)
312            .or_insert_with_key(|peer| PeerInfo {
313                supports_protocol: true,
314                dht_path: None,
315                connected: false,
316                key: peer_key(peer),
317                last_seen: now.clone(),
318                addresses: Vec::new(),
319            });
320        if *now > info.last_seen {
321            info.last_seen = now.clone();
322        }
323        info
324    }

326    /// `closest_known` paired with each peer's key, so callers that compute further distances
327    /// reuse the key instead of looking it up again.
328    fn closest_known_keyed(&self, topic: &Key, limit: usize) -> Vec<(PeerId, Key)> {
329        let mut candidates = self
330            .candidates()
331            .map(|(peer, info)| (xor_distance(topic, &info.key), peer, info.key))
332            .collect::<Vec<_>>();
333        // Only the `limit` closest peers are kept, so the rest need no sorting.
334        if limit < candidates.len() {
335            candidates.select_nth_unstable(limit);
336            candidates.truncate(limit);
337        }
338        candidates.sort_unstable();
339        candidates
340            .into_iter()
341            .map(|(_, peer, key)| (peer.clone(), key))
342            .collect()
343    }
344
345    /// Peers that may hold a topic. See the [module documentation](self).
346    fn candidates(&self) -> impl Iterator<Item = (&PeerId, &PeerInfo<TNow>)> {
347        let dht_path_known = self.dht_path_known();
348        self.discovered
349            .iter()
350            .filter(move |(_, info)| info.is_candidate(dht_path_known))
351    }
352
353    /// Whether a known statement peer speaks `statement/3`.
354    fn dht_path_known(&self) -> bool {
355        self.discovered
356            .values()
357            .any(|info| info.supports_protocol && info.dht_path == Some(true))
358    }

361impl<TNow> PeerInfo<TNow> {
362    /// Whether the peer may hold a topic, given [`PeersTopology::dht_path_known`].
363    fn is_candidate(&self, dht_path_known: bool) -> bool {
364        self.supports_protocol && !(dht_path_known && self.dht_path == Some(false))
365    }
366}

368/// The peer covering the most uncovered topics, breaking ties by the smallest distance to any
369/// topic it covers, then by the smallest peer id.
370fn best_candidate(uncovered: &[(&Key, Vec<(PeerId, Key)>)]) -> Option<PeerId> {
371    // Per peer, the number of uncovered topics it covers and its distance to the closest one.
372    let mut scores = BTreeMap::<&PeerId, (usize, Key)>::new();
373    for (topic, pool) in uncovered {
374        for (peer, key) in pool {
375            let (count, best) = scores.entry(peer).or_insert((0, [u8::MAX; 32]));
376            *count += 1;
377            *best = (*best).min(xor_distance(topic, key));
378        }
379    }
380    scores
381        .into_iter()
382        .min_by_key(|(peer, (count, best))| (Reverse(*count), *best, *peer))
383        .map(|(peer, _)| peer.clone())
384}
385
386fn pool_contains(pool: &[(PeerId, Key)], peer: &PeerId) -> bool {
387    pool.iter().any(|(candidate, _)| candidate == peer)
388}

390/// Map a peer id into the 32-byte topic key space, the same way as the full nodes.
391///
392/// The keys need not match the SHA-256 Kademlia keys because the topology never queries
393/// Kademlia by topic.
394fn peer_key(peer: &PeerId) -> Key {
395    <[u8; 32]>::try_from(blake2_rfc::blake2b::blake2b(32, &[], peer.as_bytes()).as_bytes())
396        .expect("blake2b output is 32 bytes; qed")
397}
398
399fn xor_distance(a: &Key, b: &Key) -> Key {
400    core::array::from_fn(|i| a[i] ^ b[i])
401}

431    fn dht_peer(topology: &mut PeersTopology<Duration>, peer: PeerId, now: Duration) {

436    #[test]
437    fn lifecycle_mutators_are_idempotent_and_filter_protocol_support() {
438        let mut topology = topology(2);
439        let now = Duration::ZERO;
440
441        dht_peer(&mut topology, peer(2), now);
442        dht_peer(&mut topology, peer(2), now);
443        // A discovered peer is a candidate until it is known to lack the protocol.
444        topology.on_peers_discovered([(peer(3), vec![vec![3], vec![3]])], &now);
445        assert_eq!(topology.statement_peers().count(), 2);
446        topology.on_peer_identified(peer(3), false, &now);
447
448        assert_eq!(topology.known_peers_count(), 2);
449        assert_eq!(topology.closest_known(&topic(9), 10), vec![peer(2)]);
450        assert_eq!(topology.peers_for_topics(&[topic(9)]), vec![peer(2)]);
451        assert_eq!(
452            topology.statement_peers().collect::<Vec<_>>(),
453            vec![&peer(2)]
454        );
455
456        // Addresses are deduplicated, an address reported again becomes the newest, and the
457        // oldest ones go once over the cap.
458        assert_eq!(topology.peer_addresses(&peer(3)).collect::<Vec<_>>(), [[3]]);
459        let addresses = (0..=MAX_ADDRESSES_PER_PEER as u8).map(|n| vec![n]);
460        topology.on_peers_discovered([(peer(3), addresses)], &now);
461        assert_eq!(
462            topology.peer_addresses(&peer(3)).count(),
463            MAX_ADDRESSES_PER_PEER
464        );
465        assert_eq!(topology.peer_addresses(&peer(3)).next(), Some(&[1][..]));
466        assert!(topology.peer_addresses(&peer(2)).next().is_none());

468        topology.on_peer_identified(peer(2), false, &now);
469        assert!(topology.closest_known(&topic(9), 10).is_empty());
470        assert!(topology.peers_for_topics(&[topic(9)]).is_empty());
471        assert_eq!(topology.known_peers_count(), 2);

473        // An open substream implies support, even against a later negative answer, and closing
474        // it keeps the peer known.
475        topology.on_substream_opened(peer(3), true, &now);
476        topology.on_peer_identified(peer(3), false, &now);
477        assert_eq!(topology.closest_known(&topic(9), 10), vec![peer(3)]);
478        topology.on_substream_closed(peer(3), &now);
479        assert_eq!(topology.closest_known(&topic(9), 10), vec![peer(3)]);

481    }

483    #[test]
484    fn eviction_drops_stale_peers_and_bounds_known_peers() {
485        let mut topology = topology(2);
486        let start = Duration::from_secs(1);
487
488        dht_peer(&mut topology, peer(2), start);
489        topology.on_peer_identified(peer(3), false, &start);
490        assert!(!topology.evict(&(start + PEER_STALENESS_TTL - Duration::from_secs(1))));
491        assert_eq!(topology.known_peers_count(), 2);
492
493        // Re-seeing peer 2 keeps it past the moment peer 3 goes stale. Dropping peer 3, which
494        // lacks the protocol, leaves the candidates unchanged.
495        dht_peer(&mut topology, peer(2), start + Duration::from_secs(1));
496        assert!(!topology.evict(&(start + PEER_STALENESS_TTL)));
497        assert_eq!(topology.known_peers_count(), 1);
498        assert!(topology.evict(&(start + PEER_STALENESS_TTL + Duration::from_secs(1))));
499        assert_eq!(topology.known_peers_count(), 0);
500
501        // A connected peer outlives the TTL, and an event with an older `now` leaves
502        // `last_seen` in place.
503        topology.on_substream_opened(peer(2), true, &start);
504        dht_peer(&mut topology, peer(3), start);
505        topology.on_peers_discovered([(peer(3), Vec::new())], &Duration::ZERO);
506        assert!(!topology.evict(&(Duration::ZERO + PEER_STALENESS_TTL)));
507        assert!(topology.evict(&(start + PEER_STALENESS_TTL)));
508        assert_eq!(topology.closest_known(&topic(9), 10), vec![peer(2)]);
509        topology.on_substream_closed(peer(2), &(start + PEER_STALENESS_TTL));
510        topology.on_substream_closed(peer(2), &start);
511        assert!(!topology.evict(&(start + 2 * PEER_STALENESS_TTL - Duration::from_secs(1))));
512        assert!(topology.evict(&(start + 2 * PEER_STALENESS_TTL)));
513        assert_eq!(topology.known_peers_count(), 0);
514
515        // Over the cap, the least recently seen peers go first.
516        let numbered = |n: u32| {
517            let mut key = [0; 32];
518            key[..4].copy_from_slice(&n.to_be_bytes());
519            PeerId::from_public_key(&PublicKey::Ed25519(key))
520        };
521        for n in 0..(MAX_KNOWN_PEERS + 50) as u32 {
522            dht_peer(
523                &mut topology,
524                numbered(n),
525                start + Duration::from_secs(n.into()),
526            );
527        }
528        dht_peer(
529            &mut topology,
530            numbered(0),
531            start + Duration::from_secs(1 << 20),
532        );
533        assert!(topology.evict(&start));
534        assert_eq!(topology.known_peers_count(), MAX_KNOWN_PEERS);
535        let known = topology.closest_known(&topic(9), MAX_KNOWN_PEERS);
536        assert!(known.contains(&numbered(0)));
537        assert!(!known.contains(&numbered(1)));
538        assert!(!known.contains(&numbered(50)));
539        assert!(known.contains(&numbered(51)));
540    }

622    fn routing_targets_are_connected_replicas() {
623        // Closest to topic 1 with seeds 2 to 30: 15, 7, 12, 21. Closest to topic 9: 15, 7, 21,
624        // 4. Closest to topic 17: 12, 21, 4, 15.
625        let mut topology = topology(2);

628        }
629        assert!(topology.routing_targets(&[topic(1)]).is_empty());
630
631        // Peer 21 is connected but is no replica of topic 1.
632        for seed in [7, 21, 4] {
633            topology.on_substream_opened(peer(seed), true, &Duration::ZERO);
634        }
635        assert_eq!(
636            topology.routing_targets(&[topic(1)]),
637            BTreeSet::from([peer(7)])

639        assert!(topology.routing_targets(&[]).is_empty());
640
641        // A peer close to several topics comes once.
642        assert_eq!(

664    #[test]
665    fn a_peer_without_dht_is_a_candidate_only_while_no_peer_runs_the_dht_path() {
666        // Closest to topic 1 with seeds 2 to 30: 15, 7, 12.
667        let mut topology = topology(2);
668        let now = Duration::ZERO;
669        for seed in 2..=30 {
670            topology.on_peers_discovered([(peer(seed), Vec::new())], &now);
671        }
672
673        // No peer is known to run the DHT path, so a peer on an older version stays a candidate.
674        topology.on_substream_opened(peer(15), false, &now);
675        assert_eq!(topology.closest_known(&topic(1), 2), [peer(15), peer(7)]);
676        assert_eq!(topology.statement_peers().count(), 29);
677
678        // One peer on `statement/3` is enough to drop it, and closing its substream keeps it out.
679        topology.on_substream_opened(peer(30), true, &now);
680        assert_eq!(topology.closest_known(&topic(1), 2), [peer(7), peer(12)]);
681        assert_eq!(topology.peers_for_topics(&[topic(1)]), [peer(7)]);
682        assert_eq!(topology.statement_peers().count(), 28);
683        topology.on_substream_closed(peer(15), &now);
684        assert_eq!(topology.closest_known(&topic(1), 2), [peer(7), peer(12)]);
685
686        // Identify listing `statement/3` brings it back, and a `statement/3` substream does too.
687        topology.on_peer_identified(peer(15), true, &now);
688        assert_eq!(topology.closest_known(&topic(1), 2), [peer(15), peer(7)]);
689        topology.on_substream_opened(peer(7), false, &now);
690        topology.on_peer_identified(peer(7), true, &now);
691        assert_eq!(topology.closest_known(&topic(1), 2), [peer(15), peer(12)]);
692        topology.on_substream_opened(peer(7), true, &now);
693        assert_eq!(topology.closest_known(&topic(1), 2), [peer(15), peer(7)]);
694    }

697    fn eviction_reports_only_a_change_of_the_candidates() {
698        let mut topology = topology(2);
699        let start = Duration::ZERO;
700        let stale = start + PEER_STALENESS_TTL;
701
702        // Peer 2 is on an older version and peer 3 runs the DHT path, so only peer 3 is a
703        // candidate and the eviction of peer 2 changes nothing.
704        topology.on_substream_opened(peer(2), false, &start);
705        topology.on_substream_closed(peer(2), &start);
706        topology.on_substream_opened(peer(3), true, &start);
707        assert!(!topology.evict(&stale));
708        assert_eq!(topology.known_peers_count(), 1);
709
710        // The eviction of the last peer on the DHT path makes peer 2 a candidate again.
711        topology.on_substream_opened(peer(2), false, &stale);
712        topology.on_substream_closed(peer(3), &start);
713        assert!(topology.evict(&stale));
714        assert_eq!(
715            topology.statement_peers().collect::<Vec<_>>(),
716            vec![&peer(2)]
717        );
718    }
