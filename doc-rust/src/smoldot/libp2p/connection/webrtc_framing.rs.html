29/// State of the framing.
30pub struct WebRtcFraming {
31    /// Value of [`ReadWrite::expected_incoming_bytes`] of the inner stream the last time that
32    /// [`WebRtcFraming::read_write`] was called. `None` if unknown.
33    inner_stream_expected_incoming_bytes: Option<usize>,
34
35    /// Buffer containing data from a previous frame, but that doesn't contain enough data for
36    /// the underlying substream to accept it.
37    ///
38    /// In other words, `receive_buffer.len() < inner_stream_expected_incoming_bytes`.
39    // TODO: shrink_to_fit?
40    receive_buffer: Vec<u8>,
41
42    /// State of the writing side of the remote.
43    remote_write_state: RemoteWriteState,
44
45    /// State of the local writing side.
46    local_write_state: LocalWriteState,
47}
48
49enum LocalWriteState {
50    Open,
51    FinBuffered,
52    FinAcked,
53}
54
55enum RemoteWriteState {
56    Open,
57    /// The remote has sent a `FIN` in the past. Any data in [`WebRtcFraming::receive_buffer`]
58    /// is still valid and was received before the remote writing side was closed.
59    Closed,
60    ClosedAckBuffered,
61}
62
63const RECEIVE_BUFFER_CAPACITY: usize = 2048;
64/// Maximum size in bytes of the protobuf frame surrounding the message.
65const PROTOBUF_FRAME_MAX_LEN: usize = 8; // TODO: calculate better?
66/// Size in bytes of a frame carrying a flag and no data: length prefix, flag, empty data.
67const FLAG_ONLY_FRAME_LEN: usize = 5;
68const MAX_PROTOBUF_MESSAGE_LEN: usize = 16384;
69
70impl WebRtcFraming {
71    /// Initializes a new [`WebRtcFraming`].
72    pub fn new() -> Self {
73        WebRtcFraming {
74            inner_stream_expected_incoming_bytes: None,
75            receive_buffer: Vec::with_capacity(RECEIVE_BUFFER_CAPACITY),
76            remote_write_state: RemoteWriteState::Open,
77            local_write_state: LocalWriteState::Open,
78        }
79    }
80
81    /// Feeds data coming from a socket and outputs data to write to the socket.
82    ///
83    /// Returns an object that implements `Deref<Target = ReadWrite>`. This object represents the
84    /// decrypted stream of data.
85    ///
86    /// An error is returned if the protocol is being violated by the remote, if the remote wants
87    /// to reset the substream.
88    pub fn read_write<'a, TNow: Clone>(
89        &'a mut self,
90        outer_read_write: &'a mut ReadWrite<TNow>,
91    ) -> Result<InnerReadWrite<'a, TNow>, Error> {
92        // A flag-only frame is parsed whatever the underlying substream expects: a `FIN` or
93        // `RESET_STREAM` arriving in a state that reads nothing (an outbound notifications
94        // substream, a refused inbound one, ...) must still be processed, or no `FIN_ACK` is
95        // sent and the closing goes unreported. A data frame is parsed only while the substream
96        // has room for it, so that the rest stays in the outer buffer and back-pressures the
97        // remote. A substream that asks for nothing still gets one frame at a time: states that
98        // discard what they receive must drain the remote's in-flight data to reach the `FIN`
99        // behind it. Consuming it requests the next call.
100        loop {
101            // Try to parse a frame from the incoming buffer.
102            let bytes_to_discard = {
103                // TODO: we could in theory demand from the outside just the protobuf header, and then later the data, which would save some copying but might considerably complexifies the code
104                let mut parser = nom::combinator::map_parser::<_, _, nom::error::Error<&[u8]>, _, _>(
105                    nom::multi::length_data(crate::util::leb128::nom_leb128_usize),
106                    protobuf::message_decode! {
107                        #[optional] flags = 1 => protobuf::enum_tag_decode,
108                        #[optional] message = 2 => protobuf::bytes_tag_decode,
109                    },
110                );
111                match nom::Parser::parse(&mut parser, &outer_read_write.incoming_buffer) {
112                    Ok((rest, framed_message)) => {
113                        // The remote has sent a `RESET_STREAM` flag, immediately stop with an error.
114                        // The specification mentions that the receiver may discard any data already
115                        // received, which we do.
116                        if framed_message.flags.map_or(false, |f| f == 2) {
117                            return Err(Error::RemoteResetDesired);
118                        }
119
120                        // Leave a data frame the substream has no room for in the outer buffer.
121                        let has_data = framed_message.message.is_some_and(|msg| !msg.is_empty());
122                        if has_data
123                            && self.receive_buffer.len()
124                                >= cmp::max(
125                                    self.inner_stream_expected_incoming_bytes.unwrap_or(0),
126                                    1,
127                                )
128                        {
129                            break;
130                        }
131
132                        // Some protocol check.
133                        if has_data && !matches!(self.remote_write_state, RemoteWriteState::Open) {
134                            return Err(Error::DataAfterFin);
135                        }
136
137                        // Process the `FIN_ACK` flag sent by the remote.
138                        // Note that we don't treat it as an error if the remote sends the
139                        // `FIN_ACK` flag multiple times, although this is opinionated.
140                        if framed_message.flags.map_or(false, |f| f == 3) {
141                            if matches!(self.local_write_state, LocalWriteState::Open) {
142                                return Err(Error::FinAckWithoutFin);
143                            }
144                            self.local_write_state = LocalWriteState::FinAcked;
145                        }
146
147                        // Process the `FIN` flag sent by the remote.
148                        if matches!(self.remote_write_state, RemoteWriteState::Open)
149                            && framed_message.flags.map_or(false, |f| f == 0)
150                        {
151                            self.remote_write_state = RemoteWriteState::Closed;
152                        }
153
154                        // Note that any `STOP_SENDING` flag sent by the remote is ignored.
155
156                        // Copy the message of the remote out from the incoming buffer.
157                        if let Some(message) = framed_message.message {
158                            self.receive_buffer.extend_from_slice(message);
159                        }
160
161                        // Number of bytes to discard is the size of the protobuf frame.
162                        outer_read_write.incoming_buffer.len() - rest.len()
163                    }
164                    Err(nom::Err::Incomplete(needed)) => {
165                        // Not enough data in the incoming buffer for a full frame. Requesting
166                        // more.
167                        let Some(expected_incoming_bytes) =
168                            &mut outer_read_write.expected_incoming_bytes
169                        else {
170                            // TODO: is this correct anyway? substreams are never supposed to close?
171                            return Err(Error::EofIncompleteFrame);
172                        };
173                        *expected_incoming_bytes = outer_read_write.incoming_buffer.len()
174                            + match needed {
175                                nom::Needed::Size(s) => s.get(),
176                                nom::Needed::Unknown => 1,
177                            };
178                        break;
179                    }
180                    Err(_) => {
181                        // Frame decoding error.
182                        return Err(Error::InvalidFrame);
183                    }
184                }
185            };
186
187            // Discard the frame data.
188            let _extract_result = outer_read_write.incoming_bytes_take(bytes_to_discard);
189            debug_assert!(matches!(_extract_result, Ok(Some(_))));
190        }
191
192        Ok(InnerReadWrite {
193            inner_read_write: ReadWrite {
194                now: outer_read_write.now.clone(),
195                incoming_buffer: mem::take(&mut self.receive_buffer),
196                read_bytes: 0,
197                expected_incoming_bytes: if matches!(
198                    self.remote_write_state,
199                    RemoteWriteState::Open
200                ) {
201                    Some(0)
202                } else {
203                    None
204                },
205                write_buffers: Vec::new(),
206                write_bytes_queued: 0,
207                write_bytes_queueable: if matches!(self.local_write_state, LocalWriteState::Open) {
208                    outer_read_write
209                        .write_bytes_queueable
210                        .map(|outer_writable| {
211                            cmp::min(
212                                // TODO: what if the outer maximum queueable is <= PROTOBUF_FRAME_MAX_LEN? this will never happen in practice, but in theory it could
213                                outer_writable.saturating_sub(PROTOBUF_FRAME_MAX_LEN),
214                                MAX_PROTOBUF_MESSAGE_LEN - PROTOBUF_FRAME_MAX_LEN,
215                            )
216                        })
217                } else {
218                    None
219                },
220                wake_up_after: outer_read_write.wake_up_after.clone(),
221            },
222            framing: self,
223            outer_read_write,
224        })
225    }
226}

234/// Stream of data without the frames. See [`WebRtcFraming::read_write`].
235pub struct InnerReadWrite<'a, TNow: Clone> {
236    framing: &'a mut WebRtcFraming,
237    outer_read_write: &'a mut ReadWrite<TNow>,
238    inner_read_write: ReadWrite<TNow>,
239}
240
241impl<'a, TNow: Clone> ops::Deref for InnerReadWrite<'a, TNow> {
242    type Target = ReadWrite<TNow>;
243
244    fn deref(&self) -> &Self::Target {
245        &self.inner_read_write
246    }
247}
248
249impl<'a, TNow: Clone> ops::DerefMut for InnerReadWrite<'a, TNow> {
250    fn deref_mut(&mut self) -> &mut Self::Target {
251        &mut self.inner_read_write
252    }
253}
254
255impl<'a, TNow: Clone> Drop for InnerReadWrite<'a, TNow> {
256    fn drop(&mut self) {
257        // The timer is copied back before the wake-up below, so that it can't overwrite it.
258        //
259        // The inner stream expects a call before something new arrives in two cases: it
260        // consumed bytes of `receive_buffer`, or a complete frame was withheld and it now asks
261        // for more than it holds. The second happens on every substream that reads a length
262        // prefix in one call and the body in the next. Both get an immediate wake-up. A frame
263        // was withheld if the outer demand is already met: the platform starts each call at
264        // zero, and the parsing loop raises the demand above the buffer only when the next
265        // frame is incomplete.
266        //
267        // The outer demand is left at the size of the next frame rather than raised to what the
268        // inner stream expects: a small `FIN` or `RESET_STREAM` frame must not wait until that
269        // many bytes have arrived.
270        self.outer_read_write.wake_up_after = self.inner_read_write.wake_up_after.clone();
271        let frame_withheld = self
272            .outer_read_write
273            .expected_incoming_bytes
274            .is_some_and(|expected| expected <= self.outer_read_write.incoming_buffer.len());
275        let inner_asks_for_more = self
276            .inner_read_write
277            .expected_incoming_bytes
278            .is_some_and(|expected| expected > self.inner_read_write.incoming_buffer.len());
279        if self.inner_read_write.read_bytes != 0 || (frame_withheld && inner_asks_for_more) {
280            self.outer_read_write.wake_up_asap();
281        }
282        self.framing.receive_buffer = mem::take(&mut self.inner_read_write.incoming_buffer);
283        self.framing.inner_stream_expected_incoming_bytes =
284            Some(self.inner_read_write.expected_incoming_bytes.unwrap_or(0));
285
286        // Flag to send out, if any: `FIN` once the inner stream closes its writing side,
287        // `FIN_ACK` once the remote's `FIN` has been processed. `RESET_STREAM` is never sent, as
288        // closing the substream serves the same purpose, nor is `STOP_SENDING`, which doesn't
289        // fit in our API.
290        //
291        // A frame carrying data always fits in the outer credit, as the inner stream was given
292        // that credit minus `PROTOBUF_FRAME_MAX_LEN`. A frame carrying only a flag might not,
293        // right after a frame that used the credit up. It is then sent on a later call: the
294        // inner stream asks for the close again, and `remote_write_state` stays `Closed`. The
295        // platform re-polls the stream when it reports flushed bytes, so a wake-up here would
296        // only spin until then.
297        let flag_fits = self.inner_read_write.write_bytes_queued != 0
298            || self
299                .outer_read_write
300                .write_bytes_queueable
301                .is_some_and(|queueable| queueable >= FLAG_ONLY_FRAME_LEN);
302        let flag_to_send_out: Option<u32> = if !flag_fits {
303            None
304        } else if matches!(self.framing.local_write_state, LocalWriteState::Open)
305            && self.inner_read_write.write_bytes_queueable.is_none()
306        {
307            self.framing.local_write_state = LocalWriteState::FinBuffered;
308            Some(0)
309        } else if matches!(self.framing.remote_write_state, RemoteWriteState::Closed) {
310            self.framing.remote_write_state = RemoteWriteState::ClosedAckBuffered;
311            Some(3)
312        } else {
313            None
314        };
315
316        // Write out a message only if there is anything to write.
317        // TODO: consider buffering data more before flushing, to reduce the overhead of the protobuf frame?
318        if flag_to_send_out.is_some() || self.inner_read_write.write_bytes_queued != 0 {
319            // Reserve some space in `write_buffers` to later write the message length prefix.
320            let message_length_prefix_index = self.outer_read_write.write_buffers.len();
321            self.outer_read_write
322                .write_buffers
323                .push(Vec::with_capacity(4));
324
325            // Total number of bytes written below, excluding the length prefix.
326            let mut length_prefix_value = 0;
327
328            // Write the flags, if any.
329            if let Some(flag_to_send_out) = flag_to_send_out {
330                for buffer in protobuf::uint32_tag_encode(1, flag_to_send_out) {
331                    let buffer = buffer.as_ref();
332                    length_prefix_value += buffer.len();
333                    self.outer_read_write.write_buffers.push(buffer.to_owned());
334                }
335            }
336
337            // Write the data. This consists in a protobuf tag, a length, and the data itself.
338            let data_protobuf_tag = protobuf::tag_encode(2, 2).collect::<Vec<_>>();
339            length_prefix_value += data_protobuf_tag.len();
340            self.outer_read_write.write_buffers.push(data_protobuf_tag);
341            let data_len =
342                leb128::encode_usize(self.inner_read_write.write_bytes_queued).collect::<Vec<_>>();
343            length_prefix_value += data_len.len();
344            self.outer_read_write.write_buffers.push(data_len);
345            length_prefix_value += self.inner_read_write.write_bytes_queued;
346            self.outer_read_write
347                .write_buffers
348                .extend(mem::take(&mut self.inner_read_write.write_buffers));
349
350            // Now write the length prefix.
351            let length_prefix = leb128::encode_usize(length_prefix_value).collect::<Vec<_>>();
352            let total_length = length_prefix_value + length_prefix.len();
353            debug_assert!(
354                self.inner_read_write.write_bytes_queued != 0
355                    || total_length == FLAG_ONLY_FRAME_LEN
356            );
357            self.outer_read_write.write_buffers[message_length_prefix_index] = length_prefix;
358
359            // Properly update the outer `ReadWrite`.
360            self.outer_read_write.write_bytes_queued += total_length;
361            *self
362                .outer_read_write
363                .write_bytes_queueable
364                .as_mut()
365                .unwrap() -= total_length;
366        }
367    }
368}

370/// Error while decoding data.
371#[derive(Debug, derive_more::Display, derive_more::Error)]
372pub enum Error {
373    /// The remote wants to reset the substream. This is a normal situation.
374    RemoteResetDesired,
375    /// Failed to decode the protobuf header.
376    InvalidFrame,
377    /// Remote has sent data after having sent a `FIN` flag in the past.
378    DataAfterFin,
379    /// Outer substream has closed in the middle of a frame.
380    EofIncompleteFrame,
381    /// Received a `FIN_ACK` flag without having sent a `FIN` flag.
382    FinAckWithoutFin,
383}

385#[cfg(test)]
386mod tests {
387    use super::WebRtcFraming;
388    use crate::libp2p::read_write::ReadWrite;

390    /// Builds the outer `ReadWrite`, as the platform would: `incoming` has arrived from the data
391    /// channel, and `queueable` bytes can be written to it.
392    fn outer(incoming: &[u8], queueable: usize) -> ReadWrite<u64> {
393        ReadWrite {
394            now: 5,
395            incoming_buffer: incoming.to_vec(),
396            expected_incoming_bytes: Some(0),
397            read_bytes: 0,
398            write_buffers: Vec::new(),
399            write_bytes_queued: 0,
400            write_bytes_queueable: Some(queueable),
401            wake_up_after: None,
402        }
403    }
404
405    /// The inner substream does not ask for bytes (`NotificationsOut`, `NotificationsOutClosed`,
406    /// `NotificationsInRefused`, ...). A remote `FIN` must still be parsed, the inner stream told
407    /// that the remote closed its side, and a `FIN_ACK` written.
408    #[test]
409    fn fin_parsed_while_inner_expects_no_bytes() {
410        let mut framing = WebRtcFraming::new();
411        let mut rw = outer(&[], 100);
412        // The inner substream leaves `expected_incoming_bytes` at `Some(0)`.
413        drop(framing.read_write(&mut rw).unwrap());
414
415        // Length-prefixed protobuf `Message { flag: FIN }`.
416        let mut rw = outer(&[0x02, 0x08, 0x00], 100);
417        let inner = framing.read_write(&mut rw).unwrap();
418        let inner_expected = inner.expected_incoming_bytes;
419        drop(inner);
420        assert_eq!(rw.read_bytes, 3, "FIN frame left unparsed");
421        assert_eq!(
422            inner_expected, None,
423            "inner stream not told that the remote closed"
424        );
425        // `Message { flag: FIN_ACK, message: "" }`.
426        assert_eq!(
427            rw.write_buffers.concat(),
428            [0x04, 0x08, 0x03, 0x12, 0x00],
429            "no FIN_ACK written"
430        );
431    }
432
433    /// When the inner substream consumes bytes of `receive_buffer`, `InnerReadWrite::drop`
434    /// requests an immediate wake-up of the outer `ReadWrite`. Copying the inner stream's timer
435    /// back to the outer `ReadWrite` must not overwrite that wake-up.
436    #[test]
437    fn wake_up_asap_survives_drop() {
438        let mut framing = WebRtcFraming::new();
439        // A data frame carrying "abc" already arrived before the first processing.
440        let mut rw = outer(&[0x05, 0x12, 0x03, b'a', b'b', b'c'], 100);
441        {
442            let mut inner = framing.read_write(&mut rw).unwrap();
443            // The inner substream reads part of the data and expects to be called again.
444            let _ = inner.incoming_bytes_take(3);
445        }
446        assert_eq!(rw.wake_up_after, Some(5), "wake-up lost");
447    }
448
449    /// While the inner substream waits for a large amount of data (e.g. a notification body),
450    /// the outer stream must still be asked for what the *next frame* needs, so that a small
451    /// `FIN` frame reaches the framing. Asking for the whole inner need makes the platform hold
452    /// the `FIN` until that many bytes are buffered.
453    #[test]
454    fn outer_demand_is_next_frame_not_whole_inner_need() {
455        let mut framing = WebRtcFraming::new();
456        let mut rw = outer(&[], 100);
457        {
458            let mut inner = framing.read_write(&mut rw).unwrap();
459            let _ = inner.incoming_bytes_take(1000);
460        }
461        assert!(
462            rw.expected_incoming_bytes.unwrap() <= 8,
463            "outer stream asked for {:?} bytes",
464            rw.expected_incoming_bytes
465        );
466    }
467
468    /// A frame carrying only a flag is not written while the outer credit can't hold it, as
469    /// happens right after a data frame used the credit up, and is written once it can.
470    #[test]
471    fn flag_frame_waits_for_outer_credit() {
472        let mut framing = WebRtcFraming::new();
473        // `Message { flag: FIN }` arrives while only 3 bytes can be written.
474        let mut rw = outer(&[0x02, 0x08, 0x00], 3);
475        drop(framing.read_write(&mut rw).unwrap());
476        assert!(
477            rw.write_buffers.is_empty(),
478            "FIN_ACK written without credit"
479        );
480        assert_eq!(rw.write_bytes_queueable, Some(3));
481
482        let mut rw = outer(&[], 100);
483        drop(framing.read_write(&mut rw).unwrap());
484        assert_eq!(
485            rw.write_buffers.concat(),
486            [0x04, 0x08, 0x03, 0x12, 0x00],
487            "no FIN_ACK written once there is credit"
488        );
489    }
490
491    /// A data frame the substream didn't ask for stays in the outer buffer: `receive_buffer`
492    /// holds at most one such frame. Substreams read a length prefix in one call and ask for the
493    /// body in the next, without consuming anything: the framing must then request an immediate
494    /// wake-up, as the outer demand (the next frame's size) is already met, and the next call
495    /// parses the waiting frame.
496    #[test]
497    fn unrequested_data_frame_waits_in_outer_buffer() {
498        let mut framing = WebRtcFraming::new();
499        // Data frames "abc" and "def" arrive while the substream expects no bytes.
500        let mut rw = outer(
501            &[
502                0x05, 0x12, 0x03, b'a', b'b', b'c', 0x05, 0x12, 0x03, b'd', b'e', b'f',
503            ],
504            100,
505        );
506        let inner = framing.read_write(&mut rw).unwrap();
507        assert_eq!(
508            inner.incoming_buffer, b"abc",
509            "more than one unrequested frame parsed"
510        );
511        drop(inner);
512        assert_eq!(
513            rw.read_bytes, 6,
514            "second frame not left in the outer buffer"
515        );
516        assert_eq!(rw.wake_up_after, None);
517
518        // The substream now asks for 6 bytes.
519        {
520            let mut inner = framing.read_write(&mut rw).unwrap();
521            assert!(matches!(inner.incoming_bytes_take(6), Ok(None)));
522        }
523        assert_eq!(
524            rw.wake_up_after,
525            Some(5),
526            "no wake-up for the waiting frame"
527        );
528
529        let inner = framing.read_write(&mut rw).unwrap();
530        assert_eq!(
531            inner.incoming_buffer, b"abcdef",
532            "waiting frame not parsed once asked for"
533        );
534    }

536    /// A substream that discards what it receives asks for nothing, yet must reach the `FIN`
537    /// behind the remote's in-flight data: it gets one frame per call, and parsing resumes once
538    /// it drained it.
539    #[test]
540    fn discarding_inner_reaches_fin_behind_data() {
541        let mut framing = WebRtcFraming::new();
542        // Data frames "abc" and "def", then `Message { flag: FIN }`.
543        let mut rw = outer(
544            &[
545                0x05, 0x12, 0x03, b'a', b'b', b'c', 0x05, 0x12, 0x03, b'd', b'e', b'f', 0x02, 0x08,
546                0x00,
547            ],
548            100,
549        );
550        {
551            let mut inner = framing.read_write(&mut rw).unwrap();
552            assert_eq!(inner.incoming_buffer, b"abc");
553            inner.discard_all_incoming();
554        }
555        assert_eq!(rw.wake_up_after, Some(5), "no wake-up after draining");
556
557        let mut inner = framing.read_write(&mut rw).unwrap();
558        assert_eq!(inner.incoming_buffer, b"def");
559        assert_eq!(
560            inner.expected_incoming_bytes, None,
561            "FIN behind the data not reached"
562        );
563        inner.discard_all_incoming();
564        drop(inner);
565        assert_eq!(
566            rw.write_buffers.concat(),
567            [0x04, 0x08, 0x03, 0x12, 0x00],
568            "no FIN_ACK written"
569        );
570    }
